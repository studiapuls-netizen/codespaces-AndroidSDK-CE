    private fun setupSplash() {
        val splashOverlay = findViewById<View>(R.id.splash_overlay)
        val splashRipple = findViewById<RippleView>(R.id.splash_ripple)
        val splashLogo = findViewById<ImageView>(R.id.splash_logo)

        splashLogo.scaleX = 0.7f
        splashLogo.scaleY = 0.7f
        splashLogo.alpha = 0f
        splashLogo.animate()
            .alpha(1f)
            .scaleX(1f)
            .scaleY(1f)
            .setDuration(500)
            .start()

        val rippleRun = object : Runnable {
            override fun run() {
                if (splashRipple.width > 0) {
                    splashRipple.start(
                        splashRipple.width / 2f,
                        splashRipple.height / 2f
                    )
                }
                splashHandler.postDelayed(this, SPLASH_RIPPLE_INTERVAL_MS)
            }
        }
        splashRippleRun = rippleRun
        splashRipple.post { rippleRun.run() }

        splashHandler.postDelayed({
            splashRippleRun?.let { splashHandler.removeCallbacks(it) }
            splashOverlay.animate()
                .alpha(0f)
                .withLayer()
                .setDuration(SPLASH_FADE_MS)
                .withEndAction { splashOverlay.visibility = View.GONE }
                .start()
        }, SPLASH_DURATION_MS)
    }

    private fun attachAudioBoostIfNeeded() {
        val p = globalPlayer ?: return

        if (id == 0) return
        if (id == lastAudioSessionId) return
        lastAudioSessionId = id
        audioBoost.attach(id)
        audioBoost.fadeIn()
    }

    private fun setupBottomNav(savedInstanceState: Bundle?) {
        if (savedInstanceState == null) {
            supportFragmentManager.beginTransaction()
                .replace(R.id.feed_container, FeedFragment())
                .commit()
        }

        bottomNav.setOnItemSelectedListener { item ->
            try {
                when (item.itemId) {
                    R.id.nav_home -> {
                        supportFragmentManager.beginTransaction()
                            .replace(R.id.feed_container, FeedFragment())
                            .commit()
                        true
                    }
                    R.id.nav_search -> {
                        supportFragmentManager.beginTransaction()
                            .replace(R.id.feed_container, ru.asmr1.app.ui.SearchFragment())
                            .commit()
                        true
                    }
                    R.id.nav_subs -> {
                        supportFragmentManager.beginTransaction()
                            .replace(R.id.feed_container, ru.asmr1.app.ui.SubscriptionsFragment())
                            .commit()
                        true
                    }
                    R.id.nav_bell -> {
                        supportFragmentManager.beginTransaction()
                            .replace(R.id.feed_container, StubFragment.newInstance("Уведомления — скоро"))
                            .commit()
                        true
                    }
                    R.id.nav_profile -> {
                        if (authManager.isLoggedIn()) {
                            supportFragmentManager.beginTransaction()
                                .replace(R.id.feed_container, ProfileFragment())
                                .commit()
                        } else {
                            showLoginScreen()
                        }
                        true
                    }
                    else -> false
                }
            } catch (e: Exception) {
                Toast.makeText(
                    this,
                    "CRASH: ${e.javaClass.simpleName}: ${e.message}",
                    Toast.LENGTH_LONG
                ).show()
                false
            }
        }

    fun openPlayer(video: Video) {
        currentVideo = video

        val mode = PlaybackModeResolver.resolve(this, video)
        val playUri = when (mode) {
            is PlaybackModeResolver.Mode.VideoMode -> mode.uri
            is PlaybackModeResolver.Mode.AudioMode -> mode.uri
        }

        globalPlayer?.apply {
            stop()
            clearMediaItems()
            setMediaItem(MediaItem.fromUri(playUri))
            prepare()
            playWhenReady = true
        }

        overlayContainer.visibility = View.VISIBLE
        overlayContainer.translationY = screenHeightPx
        overlayContainer.alpha = 1f

        playerContainer.visibility = View.VISIBLE
        contentContainer.visibility = View.VISIBLE
        contentOverlayContainer.visibility = View.VISIBLE
        descriptionOverlayContainer.visibility = View.GONE
        commentsOverlayContainer.visibility = View.GONE
        miniPlayerWrap.visibility = View.GONE
        miniPlayerView.player = null

        playerContainer.translationY = 0f
        playerContainer.alpha = 1f
        contentOverlayContainer.translationY = 0f
        contentOverlayContainer.alpha = 1f

        openPlayerFragments(video)

        overlayContainer.post {
            overlayContainer.animate()
                .translationY(0f)
                .setDuration(320)
                .setInterpolator(DecelerateInterpolator())
                .start()
        }

        enablePip()
    }

    fun collapseToMini() {
        if (currentVideo == null || globalPlayer == null) return
        if (isAnimatingCollapse) return
        isAnimatingCollapse = true

        descriptionOverlayContainer.visibility = View.GONE
        commentsOverlayContainer.visibility = View.GONE

        val pf = getPlayerFragmentOrNull()
        val playerFrame = pf?.view?.findViewById<View>(R.id.player_frame)
        val playerRoot = pf?.view
        val contentView = getContentFragmentOrNull()?.view

        playerRoot?.animate()?.cancel()
        playerFrame?.animate()?.cancel()
        contentView?.animate()?.cancel()

        val targetW = (200 * density).toInt()
        val targetWrapH = wrapHeightForWidth(targetW)

        val overlayW = overlayContainer.width
        val overlayH = overlayContainer.height
        val targetX = (overlayW - targetW - marginPx).toFloat()
        val targetY = (overlayH - targetWrapH - marginPx).toFloat()

        val pfLoc = IntArray(2)
        playerFrame?.getLocationInWindow(pfLoc)
        val ovLoc = IntArray(2)
        overlayContainer.getLocationInWindow(ovLoc)
        val frameX = (pfLoc[0] - ovLoc[0]).toFloat()
        val frameY = (pfLoc[1] - ovLoc[1]).toFloat()
        val frameW = (playerFrame?.width ?: 1).toFloat().coerceAtLeast(1f)
        val frameH = (playerFrame?.height ?: 1).toFloat().coerceAtLeast(1f)

        val targetCardH = (targetW * CARD_RATIO).toInt()
        val scaleX = targetW / frameW
        val scaleY = targetCardH / frameH
        val dx = targetX - frameX - (frameW * (1 - scaleX) / 2f)
        val dy = targetY - frameY - (frameH * (1 - scaleY) / 2f)

        miniPlayerView.player = globalPlayer
        val lp = miniPlayerWrap.layoutParams
        lp.width = targetW
        lp.height = targetWrapH
        miniPlayerWrap.layoutParams = lp
        miniPlayerWrap.x = targetX
        miniPlayerWrap.y = targetY
        miniPlayerWrap.alpha = 0f
        miniPlayerWrap.visibility = View.VISIBLE

        playerFrame?.animate()
            ?.scaleX(scaleX)
            ?.scaleY(scaleY)
            ?.translationX(dx)
            ?.translationY(dy)
            ?.setDuration(300)
            ?.withEndAction {
                miniPlayerWrap.alpha = 1f
                playerRoot?.visibility = View.GONE
                playerFrame?.animate()?.cancel()
                playerFrame?.scaleX = 1f
                playerFrame?.scaleY = 1f
                playerFrame?.translationX = 0f
                playerFrame?.translationY = 0f
                playerRoot?.translationY = 0f

                removePlayerFragments()
                supportFragmentManager.findFragmentById(R.id.description_overlay_container)?.let {
                    supportFragmentManager.beginTransaction().remove(it).commit()
                }
                supportFragmentManager.findFragmentById(R.id.comments_overlay_container)?.let {
                    supportFragmentManager.beginTransaction().remove(it).commit()
                }

                playerContainer.visibility = View.GONE
                contentContainer.visibility = View.GONE
                overlayContainer.visibility = View.VISIBLE
                overlayContainer.translationY = 0f

                feedContainer.visibility = View.VISIBLE
                isAnimatingCollapse = false

                bottomNav.visibility = View.VISIBLE

                refreshPipSourceRect()
            }
            ?.start()

        contentView?.animate()
            ?.translationY((contentView.height).toFloat())
            ?.alpha(0f)
            ?.setDuration(300)
            ?.start()
    }

    internal fun enablePip() {
        if (Build.VERSION.SDK_INT < Build.VERSION_CODES.S) return
        try {
            val builder = PictureInPictureParams.Builder()
                .setAspectRatio(Rational(16, 9))
                .setAutoEnterEnabled(true)
            if (miniPlayerWrap.visibility == View.VISIBLE &&
                miniPlayerContainer.width > 0 && miniPlayerContainer.height > 0) {
                val rect = Rect()
                miniPlayerContainer.getGlobalVisibleRect(rect)
                if (!rect.isEmpty) builder.setSourceRectHint(rect)
            }
            setPictureInPictureParams(builder.build())
        } catch (_: Exception) {}
    }

    internal fun disablePip() {
        if (Build.VERSION.SDK_INT < Build.VERSION_CODES.S) return
        try {
            setPictureInPictureParams(
                PictureInPictureParams.Builder().setAutoEnterEnabled(false).build()
            )
        } catch (_: Exception) {}
    }

    internal fun refreshPipSourceRect() {
        if (miniPlayerWrap.visibility != View.VISIBLE) return
        if (miniPlayerContainer.width == 0 || miniPlayerContainer.height == 0) return
        if (Build.VERSION.SDK_INT < Build.VERSION_CODES.S) return
        try {
            val rect = Rect()
            miniPlayerContainer.getGlobalVisibleRect(rect)
            if (rect.isEmpty) return

/**
 * Mini-плеер: жесты перетаскивания и pinch-to-zoom.
 *
 * Extension-функции для MainActivity.
 * Константы CARD_RATIO / CONTROLS_DP / CONTROLS_MARGIN_DP теперь top-level в MainActivity.kt.
 */

@SuppressLint("ClickableViewAccessibility")
internal fun MainActivity.setupMiniPlayerGestures() {
    var startX = 0f
    var startY = 0f
    var startTouchX = 0f
    var startTouchY = 0f
    var isDragging = false
    var downTime = 0L

    var isPinching = false
    var pinchStartDist = 0f
    var pinchStartWidth = 0
    var pinchCenterX = 0f
    var pinchCenterY = 0f

    miniPlayerWrap.setOnTouchListener { v, event ->
        val parent = v.parent as? ViewGroup ?: return@setOnTouchListener false
        val parentW = parent.width.toFloat()
        val parentH = parent.height.toFloat()
        val marginF = marginPx.toFloat()
        val minWidthPx = (minWidthDp * density).toInt()
        val maxWidthPx = (maxWidthDp * density).toInt()

        when (event.actionMasked) {
            MotionEvent.ACTION_DOWN -> {
                startX = v.x
                startY = v.y
                startTouchX = event.rawX
                startTouchY = event.rawY
                isDragging = false
                downTime = System.currentTimeMillis()
                true
            }
            MotionEvent.ACTION_POINTER_DOWN -> {
                if (event.pointerCount >= 2) {
                    isPinching = true
                    isDragging = false
                    pinchStartDist = distanceBetween(event, 0, 1)
                    pinchStartWidth = v.width
                    pinchCenterX = v.x + v.width / 2f
                    pinchCenterY = v.y + v.height / 2f
                }
                true
            }
            MotionEvent.ACTION_MOVE -> {
                if (isPinching && event.pointerCount >= 2) {
                    val curDist = distanceBetween(event, 0, 1)
                    if (pinchStartDist > 0f && pinchStartWidth > 0) {
                        val factor = curDist / pinchStartDist
                        var newW = (pinchStartWidth * factor).toInt()
                        newW = newW.coerceIn(minWidthPx, maxWidthPx)
                        val newH = wrapHeightForWidth(newW)
                        val maxX = (parentW - newW - marginPx).coerceAtLeast(marginF)
                        val maxY = (parentH - newH - marginPx).coerceAtLeast(marginF)
                        val lp = v.layoutParams
                        lp.width = newW
                        lp.height = newH
                        v.layoutParams = lp
                        v.x = (pinchCenterX - newW / 2f).coerceIn(marginF, maxX)
                        v.y = (pinchCenterY - newH / 2f).coerceIn(marginF, maxY)
                    }
                    return@setOnTouchListener true
                }
                val maxX = (parentW - v.width - marginPx).coerceAtLeast(marginF)
                val maxY = (parentH - v.height - marginPx).coerceAtLeast(marginF)
                val dx = event.rawX - startTouchX
                val dy = event.rawY - startTouchY
                if (abs(dx) > 10 || abs(dy) > 10) isDragging = true
                v.x = (startX + dx).coerceIn(marginF, maxX)
                v.y = (startY + dy).coerceIn(marginF, maxY)
                true
            }
            MotionEvent.ACTION_POINTER_UP -> {
                if (isPinching) {
                    val idx = event.actionIndex
                    val remainIdx = if (idx == 0) 1 else 0
                    if (remainIdx < event.pointerCount) {
                        startTouchX = event.getRawX(remainIdx)
                        startTouchY = event.getRawY(remainIdx)
                        startX = v.x
                        startY = v.y
                    }
                    isPinching = false
                    isDragging = false
                }
                true
            }
            MotionEvent.ACTION_UP -> {
                if (!isDragging && System.currentTimeMillis() - downTime < 250) {
                    expandFromMini()
                } else if (isDragging) {
                    val maxX = (parentW - v.width - marginPx).coerceAtLeast(marginF)
                    val maxY = (parentH - v.height - marginPx).coerceAtLeast(marginF)
                    val cx = v.x + v.width / 2f
                    val cy = v.y + v.height / 2f
                    val targetX = if (cx < parentW / 2f) marginF else maxX
                    val targetY = if (cy < parentH / 2f) marginF else maxY
                    v.animate().x(targetX).y(targetY).setDuration(200)
                        .withEndAction { refreshPipSourceRect() }.start()
                }
                isDragging = false
                isPinching = false
                refreshPipSourceRect()
                true
            }
            MotionEvent.ACTION_CANCEL -> {
                isDragging = false
                isPinching = false
                refreshPipSourceRect()
                true
            }
            else -> false
        }
    }

    val blockDrag = View.OnTouchListener { btn, ev ->
        when (ev.actionMasked) {
            MotionEvent.ACTION_DOWN -> btn.parent?.requestDisallowInterceptTouchEvent(true)
            MotionEvent.ACTION_UP, MotionEvent.ACTION_CANCEL -> btn.parent?.requestDisallowInterceptTouchEvent(false)
        }
        false
    }
    miniBtnClose.setOnTouchListener(blockDrag)
    miniBtnPrev.setOnTouchListener(blockDrag)
    miniBtnPlayPause.setOnTouchListener(blockDrag)
    miniBtnNext.setOnTouchListener(blockDrag)
}

internal fun MainActivity.wrapHeightForWidth(width: Int): Int {
    val cardH = (width * CARD_RATIO).toInt()
    val controlsH = (CONTROLS_DP * density).toInt()
    val marginH = (CONTROLS_MARGIN_DP * density).toInt()
    return cardH + marginH + controlsH
}

internal fun MainActivity.distanceBetween(event: MotionEvent, i1: Int, i2: Int): Float {
    val dx = event.getX(i1) - event.getX(i2)
    val dy = event.getY(i1) - event.getY(i2)
    return hypot(dx, dy)
}

    private fun setupTapAndSwipe() {
        var touchStartY = 0f
        var touchStartX = 0f
        var movedEnoughForSwipe = false
        var movedUpForFullscreen = false
        var movedBackSwipe = false
        var downInsidePlayer = false

        val backSwipeThresholdPx = (80 * resources.displayMetrics.density)

        binding.playerRoot.setOnTouchListener { _, event ->
            gestureDetector?.onTouchEvent(event)

            when (event.action) {
                MotionEvent.ACTION_DOWN -> {
                    touchStartY = event.rawY
                    touchStartX = event.rawX
                    movedEnoughForSwipe = false
                    movedUpForFullscreen = false
                    movedBackSwipe = false
                    downInsidePlayer = isPointInsidePlayerView(event)

                    if (downInsidePlayer) startLongPress()

                    false
                }
                MotionEvent.ACTION_MOVE -> {
                    val dy = event.rawY - touchStartY
                    val dx = event.rawX - touchStartX

                    if (abs(dy) > 20 || abs(dx) > 20) {
                        stopLongPress()
                    }

                    if (isFullscreen && dx > backSwipeThresholdPx && dx > abs(dy) * 1.5f) {
                        movedBackSwipe = true
                        return@setOnTouchListener true
                    }

                    if (dy < -50 && abs(dy) > abs(dx) * 1.5f && !isFullscreen) {
                        movedUpForFullscreen = true
                    }

                    if (dy > 30 && dy > abs(dx) * 1.5f) {
                        movedEnoughForSwipe = true
                        (requireActivity() as MainActivity).onPlayerSwipeDown(dy)
                    }

                    movedEnoughForSwipe || movedUpForFullscreen || movedBackSwipe
                }
                MotionEvent.ACTION_UP -> {
                    stopLongPress()

                    if (movedBackSwipe) {
                        handleBack()
                        true
                    } else if (movedUpForFullscreen && !isFullscreen) {
                        enterFullscreen()
                        true
                    } else if (movedEnoughForSwipe) {
                        val deltaY = event.rawY - touchStartY
                        if (deltaY > 150) {
                            val activity = requireActivity() as MainActivity
                            if (isFullscreen) {
                                exitFullscreen()
                                binding.playerView.postDelayed({
                                    activity.collapseToMini()
                                }, 250)
                            } else {
                                activity.collapseToMini()
                            }
                        } else {
                            (requireActivity() as MainActivity).onPlayerSwipeDown(0f)
                        }
                        true
                    } else {
                        false
                    }
                }
                MotionEvent.ACTION_CANCEL -> {
                    stopLongPress()
                    false
                }
                else -> false
            }
        }
    }

    fun handleBack(): Boolean {
        if (isFullscreen) {
            exitFullscreen()
            return true
        }
        return false
    }

                )

        val runnable = object : Runnable {
            override fun run() {
                if (_binding != null) {
                    updateTimes()
                    handler.postDelayed(this, 500)
                }
            }
        }
        handler.post(runnable)

        binding.btnPlayPause.setImageResource(
            if (player.isPlaying) R.drawable.ic_player_pause else R.drawable.ic_player_play
        )
        updateTimes()
    }

    private fun updateTimes() {
        if (_binding == null) return
        if (userSeeking) return
        val player = MainActivity.globalPlayer ?: return

        val duration = player.duration.coerceAtLeast(0L)
        val position = player.currentPosition.coerceAtLeast(0L)

        binding.currentTime.text = formatTime(position)
        binding.totalTime.text = formatTime(duration)

        if (duration > 0) {
            binding.progressBar.setProgress(position.toFloat() / duration.toFloat())
            binding.progressBar.setBuffered(player.bufferedPosition.toFloat() / duration.toFloat())
        } else {
            binding.progressBar.setProgress(0f)
            binding.progressBar.setBuffered(0f)
        }
    }

    private fun formatTime(ms: Long): String {
        if (ms <= 0) return "00:00"
        val totalSec = ms / 1000
        val h = totalSec / 3600
        val m = (totalSec % 3600) / 60
        val s = totalSec % 60
        return if (h > 0) {
            String.format("%d:%02d:%02d", h, m, s)
        } else {
            String.format("%02d:%02d", m, s)
        }
    }

            try {
                val token = authManager.getToken() ?: return@launch
                val response = ApiClient.createJwtService(token).getSubscriptionStatus(authorId)
                if (_binding == null) return@launch
                if (response.success) {
                    subscribersCount = response.subscribersCount
                    subscriptionState = when {
                        !response.subscribed -> 0
                        response.notificationsEnabled -> 1
                        else -> 2
                    }
                    updateSubscriptionUi()
                }
            } catch (_: Exception) { }
        }
    }

    private fun performSubscribe(action: String) {
        if (subscribeInFlight) return
        if (!authManager.isLoggedIn()) {
            Toast.makeText(requireContext(), "Войдите чтобы подписаться", Toast.LENGTH_SHORT).show()
            return
        }
        val authorId = video?.author?.id ?: return
        subscribeInFlight = true
        val oldState = subscriptionState
        val oldCount = subscribersCount
        when (action) {
            "sub" -> { subscriptionState = 1; subscribersCount = oldCount + 1 }
            "unsub" -> { subscriptionState = 0; subscribersCount = (oldCount - 1).coerceAtLeast(0) }
            "notify" -> subscriptionState = 1
            "mute" -> subscriptionState = 2
        }
        updateSubscriptionUi()
        viewLifecycleOwner.lifecycleScope.launch {
            try {
                val token = authManager.getToken() ?: return@launch
                val response = ApiClient.createJwtService(token)
                    .subscribe(authorId, SubscribeRequest(action))
                if (_binding == null) return@launch
                if (response.success) {
                    subscribersCount = response.subscribersCount
                    subscriptionState = when {
                        !response.subscribed -> 0
                        response.notificationsEnabled -> 1
                        else -> 2
                    }
                    updateSubscriptionUi()
                } else {
                    subscriptionState = oldState
                    subscribersCount = oldCount
                    updateSubscriptionUi()
                    Toast.makeText(requireContext(), response.message ?: "Ошибка", Toast.LENGTH_SHORT).show()
                }
            } catch (e: Exception) {
                if (_binding == null) return@launch
                subscriptionState = oldState
                subscribersCount = oldCount
                updateSubscriptionUi()
                Toast.makeText(requireContext(), "Ошибка сети", Toast.LENGTH_SHORT).show()
            } finally {
                subscribeInFlight = false
            }
        }
    }

    private fun updateSubscriptionUi() {
        _binding?.apply {
            when (subscriptionState) {
                0 -> {
                    btnSubscribe.visibility = View.VISIBLE
                    btnBell.visibility = View.GONE
                }
                else -> {
                    btnSubscribe.visibility = View.GONE
                    btnBell.visibility = View.VISIBLE
                    bellIcon.setImageResource(
                        if (subscriptionState == 1) R.drawable.ic_nav_bell
                        else R.drawable.ic_bell_off
                    )
                }
            }
            authorSubs.text = "$subscribersCount подписчиков"
        }
    }

package ru.asmr1.app.player

import android.content.Context
import android.graphics.Canvas
import android.graphics.Color
import android.graphics.Paint
import android.graphics.RectF
import android.util.AttributeSet
import android.view.MotionEvent
import android.view.View

/**
 * Только полоса прогресса. Thumb рисуется отдельным View поверх (thumb_view).
 * Сообщает координаты thumb через [onThumbPosition].
 */
class VideoProgressBar @JvmOverloads constructor(
    context: Context,
    attrs: AttributeSet? = null,
    defStyleAttr: Int = 0
) : View(context, attrs, defStyleAttr) {

    private var progress = 0f
    private var buffered = 0f
    private var expanded = false
    private var cornersEnabled = false

    private val thinHeightPx = dpToPx(2f)
    private val thickHeightPx = dpToPx(3.5f)
    private val cornerPx = dpToPx(2f)

    // Высота зоны кликов
    private val touchAreaPx = dpToPx(24f)

    private val bgColor = Color.parseColor("#66000000")
    private val progressColor = Color.WHITE

    private val bgPaint = Paint(Paint.ANTI_ALIAS_FLAG).apply {
        color = bgColor
        style = Paint.Style.FILL
    }
    private val progressPaint = Paint(Paint.ANTI_ALIAS_FLAG).apply {
        color = progressColor
        style = Paint.Style.FILL
    }

    private val tmpRect = RectF()

    var onSeek: ((Float) -> Unit)? = null
    var onSeekStart: (() -> Unit)? = null
    var onSeekEnd: (() -> Unit)? = null

    /** Callback для перемещения внешнего thumb: (x, y) в координатах этого View. */
    var onThumbPosition: ((Float, Float) -> Unit)? = null

    private var isDragging = false

    fun setProgress(p: Float) {
        progress = p.coerceIn(0f, 1f)
        invalidate()
        notifyThumbPosition()
    }

    fun setBuffered(b: Float) {
        buffered = b.coerceIn(0f, 1f)
        invalidate()
    }

    fun setExpanded(expand: Boolean) {
        expanded = expand
        invalidate()
    }

    fun setCornersEnabled(enabled: Boolean) {
        if (cornersEnabled == enabled) return
        cornersEnabled = enabled
        invalidate()
    }

    override fun onMeasure(widthMeasureSpec: Int, heightMeasureSpec: Int) {
        val desiredHeight = touchAreaPx.toInt()
        val h = resolveSize(desiredHeight, heightMeasureSpec)
        val w = resolveSize(suggestedMinimumWidth, widthMeasureSpec)
        setMeasuredDimension(w, h)
    }

    override fun onDraw(canvas: Canvas) {
        super.onDraw(canvas)

        val w = width.toFloat()
        val h = height.toFloat()

        val barH = if (expanded) thickHeightPx else thinHeightPx

        // Полоса в самом низу View
        val barBottom = h
        val barTop = h - barH

        val r = if (cornersEnabled) cornerPx else 0f

        // Фон полосы
        tmpRect.set(0f, barTop, w, barBottom)
        canvas.drawRoundRect(tmpRect, r, r, bgPaint)

        // Прогресс
        val progressW = w * progress
        if (progressW > 0f) {
            tmpRect.set(0f, barTop, progressW, barBottom)
            canvas.drawRoundRect(tmpRect, r, r, progressPaint)
        }

        // Thumb рисуется ВНЕШНИМ View, но координаты сообщаем отсюда
        notifyThumbPosition()
    }

    private fun notifyThumbPosition() {
        if (width == 0 || height == 0) return
        val barH = if (expanded) thickHeightPx else thinHeightPx
        val barBottom = height.toFloat()
        val barTop = barBottom - barH
        val cx = (width * progress).coerceIn(0f, width.toFloat())
        // Центр thumb на полосе
        val cy = barTop + barH / 2f
        onThumbPosition?.invoke(cx, cy)
    }

    override fun onTouchEvent(event: MotionEvent): Boolean {
        when (event.actionMasked) {
            MotionEvent.ACTION_DOWN -> {
                isDragging = true
                parent?.requestDisallowInterceptTouchEvent(true)
                onSeekStart?.invoke()
                handleTouch(event.x)
                return true
            }
            MotionEvent.ACTION_MOVE -> {
                if (isDragging) {
                    handleTouch(event.x)
                }
                return true
            }
            MotionEvent.ACTION_UP, MotionEvent.ACTION_CANCEL -> {
                if (isDragging) {
                    isDragging = false
                    parent?.requestDisallowInterceptTouchEvent(false)
                    onSeekEnd?.invoke()
                }
                return true
            }
        }
        return super.onTouchEvent(event)
    }

    private fun handleTouch(x: Float) {
        val p = (x / width).coerceIn(0f, 1f)
        progress = p
        invalidate()
        notifyThumbPosition()
        onSeek?.invoke(p)
    }

    fun setDragging(dragging: Boolean) {
        isDragging = dragging
    }

    private fun dpToPx(dp: Float): Float =
        dp * resources.displayMetrics.density
}

 */
class RippleView @JvmOverloads constructor(
    context: Context,
    attrs: AttributeSet? = null,
    defStyleAttr: Int = 0
) : View(context, attrs, defStyleAttr) {

    private var centerX = 0f
    private var centerY = 0f
    private var radius = 0f
    private var maxRadius = 0f
    private var alpha = 0f

    private val paint = Paint(Paint.ANTI_ALIAS_FLAG).apply {
        style = Paint.Style.FILL
        color = Color.WHITE
    }

    private var animator: ValueAnimator? = null

    fun start(touchX: Float, touchY: Float) {
        centerX = touchX
        centerY = touchY
        maxRadius = Math.min(width.toFloat(), height.toFloat()) * 0.6f
        radius = 0f
        alpha = 0.35f

        animator?.cancel()
        animator = ValueAnimator.ofFloat(0f, 1f).apply {
            duration = 800
            interpolator = AccelerateDecelerateInterpolator()
            addUpdateListener { anim ->
                val v = anim.animatedValue as Float
                radius = maxRadius * v
                alpha = 0.35f * (1f - v)
                invalidate()
            }
            start()
        }
    }

    override fun onDraw(canvas: Canvas) {
        super.onDraw(canvas)
        if (radius <= 0f || alpha <= 0f) return
        paint.alpha = (alpha * 255).toInt().coerceIn(0, 255)
        canvas.drawCircle(centerX, centerY, radius, paint)
    }

    override fun onDetachedFromWindow() {
        super.onDetachedFromWindow()
        animator?.cancel()
        animator = null
    }
}

package ru.asmr1.app.player

import android.media.audiofx.LoudnessEnhancer
import android.os.Handler
import android.os.Looper

/**
 * Обёртка над LoudnessEnhancer.
 * - Усиливает звук до [targetGainMb] миллибел.
 * - Плавный fade-in с 0 до целевого усиления за [FADE_IN_MS] мс.
 */
class AudioBoost {

    companion object {
        // ×3 по амплитуде ≈ +9.5 дБ ≈ 950 mB
        private const val TARGET_GAIN_MB = 950
        private const val FADE_IN_MS = 1500L
        private const val FADE_STEP_MS = 30L
    }

    private var enhancer: LoudnessEnhancer? = null
    private var currentGainMb = 0
    private var fadeRunnable: Runnable? = null
    private val handler = Handler(Looper.getMainLooper())

    private var targetAudioSessionId: Int = 0

    fun attach(audioSessionId: Int) {
        if (audioSessionId == 0) return
        if (enhancer != null && targetAudioSessionId == audioSessionId) return

        release()
        targetAudioSessionId = audioSessionId

        try {
            enhancer = LoudnessEnhancer(audioSessionId).apply {
                setTargetGain(0)
                enabled = true
            }
        } catch (_: Exception) {
            enhancer = null
        }
    }

    /** Плавно увеличить громкость с 0 до TARGET_GAIN_MB за FADE_IN_MS. */
    fun fadeIn() {
        cancelFade()
        val e = enhancer ?: return

        currentGainMb = 0
        try {
            e.setTargetGain(0)
        } catch (_: Exception) {
        }

        val steps = (FADE_IN_MS / FADE_STEP_MS).toInt().coerceAtLeast(1)
        val stepGain = TARGET_GAIN_MB / steps
        var step = 0

        val r = object : Runnable {
            override fun run() {
                val e2 = enhancer ?: return
                step++
                currentGainMb = (currentGainMb + stepGain).coerceAtMost(TARGET_GAIN_MB)
                try {
                    e2.setTargetGain(currentGainMb)
                } catch (_: Exception) {
                }
                if (step < steps) {
                    handler.postDelayed(this, FADE_STEP_MS)
                }
            }
        }
        fadeRunnable = r
        handler.postDelayed(r, FADE_STEP_MS)
    }

    fun setGain(gainMb: Int) {
        cancelFade()
        val e = enhancer ?: return
        currentGainMb = gainMb.coerceIn(0, TARGET_GAIN_MB)
        try {
            e.setTargetGain(currentGainMb)
        } catch (_: Exception) {
        }
    }

    private fun cancelFade() {
        fadeRunnable?.let { handler.removeCallbacks(it) }
        fadeRunnable = null
    }

    fun release() {
        cancelFade()
        try {
            enhancer?.enabled = false
            enhancer?.release()
        } catch (_: Exception) {
        }
        enhancer = null
        currentGainMb = 0
        targetAudioSessionId = 0
    }
}

import android.view.LayoutInflater

import java.text.SimpleDateFormat
import java.util.Locale
import java.util.TimeZone

class VideoAdapter(
    private var videos: List<Video> = emptyList(),
    private val onClick: (Video) -> Unit = {},
    private val onMoreClick: (Video, View) -> Unit = { _, _ -> }
) : RecyclerView.Adapter<RecyclerView.ViewHolder>() {

    companion object {
        private const val TYPE_VIDEO = 0
        private const val TYPE_SKELETON = 1
        private const val SKELETON_COUNT = 5
        private const val PREFETCH_AHEAD = 2

        private val dateFmt = SimpleDateFormat("yyyy-MM-dd HH:mm:ss", Locale.US).apply {
            timeZone = TimeZone.getTimeZone("UTC")
        }
    }

    private var showSkeleton = false

    fun setVideos(newVideos: List<Video>) {
        val wasSkeleton = showSkeleton
        showSkeleton = false

        if (wasSkeleton) {
            videos = newVideos
            notifyDataSetChanged()
            return
        }

        val old = videos
        val diff = DiffUtil.calculateDiff(object : DiffUtil.Callback() {
            override fun getOldListSize() = old.size
            override fun getNewListSize() = newVideos.size
            override fun areItemsTheSame(o: Int, n: Int) = old[o].id == newVideos[n].id
            override fun areContentsTheSame(o: Int, n: Int) = old[o] == newVideos[n]
        })
        videos = newVideos
        diff.dispatchUpdatesTo(this)
    }

    fun appendVideos(newVideos: List<Video>) {
        if (newVideos.isEmpty()) return
        val startPos = videos.size
        videos = videos + newVideos
        notifyItemRangeInserted(startPos, newVideos.size)
    }

    fun showSkeleton() {
        showSkeleton = true
        videos = emptyList()
        notifyDataSetChanged()
    }

    override fun getItemViewType(position: Int): Int =
        if (showSkeleton) TYPE_SKELETON else TYPE_VIDEO

    override fun getItemCount(): Int =
        if (showSkeleton) SKELETON_COUNT else videos.size

    override fun onCreateViewHolder(parent: ViewGroup, viewType: Int): RecyclerView.ViewHolder {
        val inflater = LayoutInflater.from(parent.context)
        return when (viewType) {
            TYPE_SKELETON -> SkeletonViewHolder(
                ItemVideoSkeletonBinding.inflate(inflater, parent, false)
            )
            else -> VideoViewHolder(
                ItemVideoBinding.inflate(inflater, parent, false)
            )
        }
    }

    override fun onBindViewHolder(holder: RecyclerView.ViewHolder, position: Int) {
        when (holder) {
            is VideoViewHolder -> holder.bind(videos[position])
            is SkeletonViewHolder -> holder.bind()
        }
    }

    fun prefetchAround(context: android.content.Context, anchorPosition: Int) {
        if (showSkeleton) return
        val loader = context.imageLoader
        for (i in 1..PREFETCH_AHEAD) {
            val idx = anchorPosition + i
            if (idx >= videos.size) break
            val v = videos[idx]
            if (v.thumbnail.isNotEmpty()) {
                loader.enqueue(
                    ImageRequest.Builder(context)
                        .data(v.thumbnail)
                        .memoryCachePolicy(CachePolicy.ENABLED)
                        .diskCachePolicy(CachePolicy.ENABLED)
                        .build()
                )
            }
        }
    }

    inner class VideoViewHolder(private val binding: ItemVideoBinding) :
        RecyclerView.ViewHolder(binding.root) {

        fun bind(video: Video) {
            binding.title.text = video.title
            binding.meta.text = video.author.name

            binding.viewsOverlay.text = formatViewsShort(video.views)
            binding.dateOverlay.text = timeAgoShort(video.published)

            binding.thumbnail.transitionName = "video_thumb_" + video.id

            if (video.thumbnail.isNotEmpty()) {
                binding.thumbnail.load(video.thumbnail) {
                    crossfade(false)
                    placeholder(R.color.card_bg)
                    error(R.color.card_bg)
                    memoryCachePolicy(CachePolicy.ENABLED)
                    diskCachePolicy(CachePolicy.ENABLED)
                    allowHardware(true)
                }
            }

            if (video.author.avatar.isNotEmpty()) {
                binding.authorAvatar.load(video.author.avatar) {
                    crossfade(false)
                    placeholder(R.color.card_bg)
                    error(R.color.card_bg)
                    memoryCachePolicy(CachePolicy.ENABLED)
                    diskCachePolicy(CachePolicy.ENABLED)
                    allowHardware(true)
                }
            }

            if (video.duration > 0) {
                binding.duration.visibility = View.VISIBLE
                binding.duration.text = formatDuration(video.duration)
            } else {
                binding.duration.visibility = View.GONE
            }

            binding.root.setOnClickListener {
                val rect = android.graphics.Rect()
                binding.thumbnail.getGlobalVisibleRect(rect)
                onClick(video)
            }
            binding.btnMore.setOnClickListener { v -> onMoreClick(video, v) }
        }
    }

    class SkeletonViewHolder(private val binding: ItemVideoSkeletonBinding) :
        RecyclerView.ViewHolder(binding.root) {

        fun bind() {
            binding.veilLayout.veil()
        }
    }

    private fun formatViewsShort(n: Int): String = when {
        n >= 1_000_000 -> String.format(Locale.US, "%.1fM", n / 1_000_000.0)
        n >= 1_000 -> String.format(Locale.US, "%.1fK", n / 1_000.0)
        else -> n.toString()
    }

    private fun timeAgoShort(dateStr: String): String {
        if (dateStr.isEmpty()) return ""
        return try {
            val date = synchronized(dateFmt) { dateFmt.parse(dateStr) } ?: return ""
            val diff = (System.currentTimeMillis() - date.time) / 1000
            when {
                diff < 60 -> "сейчас"
                diff < 3600 -> "${diff / 60}м"
                diff < 86400 -> "${diff / 3600}ч"
                diff < 2_592_000 -> "${diff / 86400}д"
                diff < 31_536_000 -> "${diff / 2_592_000}мес"
                else -> "${diff / 31_536_000}г"
            }
        } catch (_: Exception) {
            ""
        }
    }
}

class VideoListFragment : Fragment() {

    companion object {
        const val ARG_TAB = "tab"
        val sharedPool: RecyclerView.RecycledViewPool by lazy { RecyclerView.RecycledViewPool() }
        fun newInstance(tab: String) = VideoListFragment().apply {
            arguments = Bundle().apply { putString(ARG_TAB, tab) }
        }
    }

    private var _binding: FragmentVideoListBinding? = null
    private val binding get() = _binding!!

    private lateinit var adapter: VideoAdapter
    private lateinit var layoutManager: LinearLayoutManager
    private var parentPager: ViewPager2? = null

    private var currentTab = "recommended"
    private var currentPage = 1
    private var isLoading = false
    private var hasMore = true
    private val perPage = 10
    private var loadJob: Job? = null

    override fun onCreateView(
        inflater: LayoutInflater,
        container: ViewGroup?,
        savedInstanceState: Bundle?
    ): View {
        _binding = FragmentVideoListBinding.inflate(inflater, container, false)
        return binding.root
    }

    override fun onViewCreated(view: View, savedInstanceState: Bundle?) {
        super.onViewCreated(view, savedInstanceState)

        currentTab = arguments?.getString(ARG_TAB) ?: "recommended"
        parentPager = findParentViewPager2(view)

        setupRecyclerView()
        setupSwipeRefresh()
        setupInfiniteScroll()
        loadFeed(1, false)
    }

    private fun findParentViewPager2(view: View): ViewPager2? {
        var v: View? = view.parent as? View
        while (v != null) {
            if (v is ViewPager2) return v
            v = v.parent as? View
        }
        return null
    }

    private fun setupRecyclerView() {
        adapter = VideoAdapter(
            onClick = { video ->
                (requireActivity() as ru.asmr1.app.MainActivity).openPlayer(video)
            },
            onMoreClick = { video, _ -> showVideoMenu(video) }
        )
        layoutManager = LinearLayoutManager(requireContext())
        binding.recyclerView.layoutManager = layoutManager
        binding.recyclerView.adapter = adapter

        binding.recyclerView.setHasFixedSize(false)
        binding.recyclerView.itemAnimator = null
        binding.recyclerView.setItemViewCacheSize(30)
        binding.recyclerView.setRecycledViewPool(sharedPool)
        binding.recyclerView.isNestedScrollingEnabled = true
        layoutManager.isItemPrefetchEnabled = true
        layoutManager.initialPrefetchItemCount = 10

        binding.recyclerView.addOnScrollListener(object : RecyclerView.OnScrollListener() {
            override fun onScrollStateChanged(recyclerView: RecyclerView, newState: Int) {
                when (newState) {
                    RecyclerView.SCROLL_STATE_DRAGGING,
                    RecyclerView.SCROLL_STATE_SETTLING -> {
                        parentPager?.isUserInputEnabled = false
                    }
                    RecyclerView.SCROLL_STATE_IDLE -> {
                        parentPager?.isUserInputEnabled = true
                    }
                }
            }
        })
    }

    fun scrollToTop() {
        _binding?.recyclerView?.scrollToPosition(0)
    }

    private fun showVideoMenu(video: Video) {
        try {
            val sheetView = layoutInflater.inflate(R.layout.sheet_video_menu, null)
            val dialog = BottomSheetDialog(requireContext())
            dialog.setContentView(sheetView)
            sheetView.findViewById<View>(R.id.menu_hide)?.setOnClickListener { dialog.dismiss() }
            sheetView.findViewById<View>(R.id.menu_hide_channel)?.setOnClickListener { dialog.dismiss() }
            sheetView.findViewById<View>(R.id.menu_save)?.setOnClickListener { dialog.dismiss() }
            sheetView.findViewById<View>(R.id.menu_report)?.setOnClickListener { dialog.dismiss() }
            sheetView.findViewById<View>(R.id.menu_cancel)?.setOnClickListener { dialog.dismiss() }
            dialog.show()
        } catch (e: Exception) {
            Log.e("VideoListFragment", "showVideoMenu failed", e)
        }
    }

    private fun setupSwipeRefresh() {
        binding.swipeRefresh.setOnRefreshListener {
            currentPage = 1
            hasMore = true
            loadFeed(1, false)
        }
    }

    private fun setupInfiniteScroll() {
        binding.recyclerView.addOnScrollListener(object : RecyclerView.OnScrollListener() {
            override fun onScrolled(recyclerView: RecyclerView, dx: Int, dy: Int) {
                super.onScrolled(recyclerView, dx, dy)
                val last = layoutManager.findLastVisibleItemPosition()
                if (last > 0) adapter.prefetchAround(recyclerView.context, last)
                if (dy <= 0) return
                if (isLoading || !hasMore) return
                if (_binding == null) return

                val totalItems = layoutManager.itemCount
                val lastVisible = layoutManager.findLastVisibleItemPosition()
                if (totalItems > 0 && lastVisible >= totalItems - 10) {
                    currentPage++
                    loadFeed(currentPage, true)
                }
            }
        })
    }

    private fun loadFeed(page: Int, append: Boolean) {
        if (isLoading) return
        isLoading = true
        if (!append) {
            if (_binding != null) {
                if (adapter.itemCount == 0) adapter.showSkeleton()
                else binding.swipeRefresh.isRefreshing = true
            }
        }
        loadJob?.cancel()
        loadJob = lifecycleScope.launch {
            try {
                val response = ApiClient.service.getFeed(tab = currentTab, page = page, perPage = perPage)
                if (_binding == null) return@launch
                if (response.success) {
                    if (append) adapter.appendVideos(response.videos)
                    else adapter.setVideos(response.videos)
                    hasMore = response.hasMore
                } else {
                    adapter.setVideos(emptyList())
                    hasMore = false
                }
            } catch (e: Exception) {
                Log.e("VideoListFragment", "Ошибка загрузки", e)
                if (_binding != null) adapter.setVideos(emptyList())
                hasMore = false
            } finally {
                isLoading = false
                _binding?.swipeRefresh?.isRefreshing = false
            }
        }
    }

    override fun onDestroyView() {
        super.onDestroyView()
        loadJob?.cancel()
        parentPager?.isUserInputEnabled = true
        _binding = null
    }
}

package ru.asmr1.app.ui

import android.content.Context
import android.content.SharedPreferences
import android.os.Bundle
import android.view.LayoutInflater
import android.view.View
import android.view.ViewGroup
import androidx.fragment.app.Fragment
import androidx.viewpager2.adapter.FragmentStateAdapter
import androidx.viewpager2.widget.ViewPager2
import ru.asmr1.app.databinding.FragmentFeedBinding
import ru.asmr1.app.download.DownloadStorage
import ru.asmr1.app.utils.NetworkUtils

class FeedFragment : Fragment() {

    private var _binding: FragmentFeedBinding? = null
    private val binding get() = _binding!!

    // Динамический список табов: "recommended" | "latest" | "downloads"
    private val tabs = mutableListOf<String>()

    // Индекс таба «Скачано» (-1 если его нет)
    private var downloadsTabIndex = -1

    // ===== Слушатель изменений в SharedPreferences («asmr1_downloads») =====
    private var prefsListener: SharedPreferences.OnSharedPreferenceChangeListener? = null

    override fun onCreateView(
        inflater: LayoutInflater,
        container: ViewGroup?,
        savedInstanceState: Bundle?
    ): View {
        _binding = FragmentFeedBinding.inflate(inflater, container, false)
        return binding.root
    }

    override fun onViewCreated(view: View, savedInstanceState: Bundle?) {
        super.onViewCreated(view, savedInstanceState)

        // ===== СОБИРАЕМ СПИСОК ТАБОВ =====
        buildTabsList()

        // ===== НАСТРАИВАЕМ ViewPager2 =====
        setupViewPager()

        // ===== КЛИКИ ПО ТАБАМ =====
        binding.tabRecommended.setOnClickListener { scrollToTab("recommended") }
        binding.tabLatest.setOnClickListener { scrollToTab("latest") }
        binding.tabDownloads.setOnClickListener { scrollToTab("downloads") }

        // ===== УПРАВЛЯЕМ ВИДИМОСТЬЮ ТАБА «СКАЧАНО» =====
        updateDownloadsTabVisibility()

        // ===== СТАРТОВАЯ СТРАНИЦА =====
        val initialIndex = if (downloadsTabIndex >= 0 &&
            !NetworkUtils.isOnline(requireContext())
        ) {
            // Офлайн → сразу на «Скачано»
            downloadsTabIndex
        } else {
            // Онлайн → на первый таб (В тренде)
            0
        }

        binding.viewPager.setCurrentItem(initialIndex, false)
        applyTabStates(initialIndex.toFloat())

        // ===== СЛУШАЕМ ИЗМЕНЕНИЯ =====
        subscribeToDownloadChanges()
    }

    /**
     * Настраивает ViewPager2 с адаптером на основе текущего списка tabs.
     * Вызывается и при первом старте, и при пересборке адаптера.
     */
    private fun setupViewPager() {
        binding.viewPager.offscreenPageLimit = 1
        binding.viewPager.adapter = object : FragmentStateAdapter(this) {
            override fun getItemCount(): Int = tabs.size
            override fun createFragment(position: Int): Fragment {
                return when (tabs.getOrNull(position)) {
                    "downloads" -> DownloadsFragment()
                    "latest" -> VideoListFragment.newInstance("latest")
                    else -> VideoListFragment.newInstance("recommended")
                }
            }
        }

        binding.viewPager.registerOnPageChangeCallback(object : ViewPager2.OnPageChangeCallback() {
            override fun onPageScrolled(
                position: Int,
                positionOffset: Float,
                positionOffsetPixels: Int
            ) {
                applyTabStates(position + positionOffset)
            }

            override fun onPageSelected(position: Int) {
                for (i in tabs.indices) {
                    if (i != position) {
                        scrollToTopAt(i)
                    }
                }
            }
        })
    }

    /**
     * Подписка на изменения в prefs "asmr1_downloads".
     * Когда что-то скачано/удалено — пересобираем табы.
     */
    private fun subscribeToDownloadChanges() {
        val prefs = requireContext()
            .getSharedPreferences("asmr1_downloads", Context.MODE_PRIVATE)

        prefsListener = SharedPreferences.OnSharedPreferenceChangeListener { _, key ->
            if (key == "downloads_index") {
                // Что-то изменилось в списке загрузок
                refreshTabsIfNeeded()
            }
        }

        prefs.registerOnSharedPreferenceChangeListener(prefsListener)
    }

    /**
     * Пересобирает список табов. Если состав изменился —
     * пересоздаёт адаптер, чтобы ViewPager2 подхватил новый таб.
     */
    private fun refreshTabsIfNeeded() {
        if (_binding == null) return

        val oldIndex = downloadsTabIndex
        val oldCount = tabs.size

        buildTabsList()
        updateDownloadsTabVisibility()

        if (tabs.size != oldCount || downloadsTabIndex != oldIndex) {
            // Количество табов изменилось — пересоздаём адаптер
            setupViewPager()
            // Возвращаемся на первый таб (или сохраняем текущую позицию если возможно)
            val newIndex = if (downloadsTabIndex >= 0 &&
                !NetworkUtils.isOnline(requireContext())
            ) {
                downloadsTabIndex
            } else {
                0
            }
            binding.viewPager.setCurrentItem(newIndex, false)
            applyTabStates(newIndex.toFloat())
        }
    }

    /**
     * Собирает список табов на основе состояния.
     */
    private fun buildTabsList() {
        tabs.clear()
        downloadsTabIndex = -1

        val storage = DownloadStorage(requireContext())
        val hasDownloads = storage.hasAnyDownloads()
        val isOnline = NetworkUtils.isOnline(requireContext())

        tabs.add("recommended")
        tabs.add("latest")

        if (hasDownloads || !isOnline) {
            tabs.add("downloads")
            downloadsTabIndex = 2
        }
    }

    private fun updateDownloadsTabVisibility() {
        binding.tabDownloads.visibility = if (downloadsTabIndex >= 0) {
            View.VISIBLE
        } else {
            View.GONE
        }
    }

    private fun scrollToTab(name: String) {
        val index = tabs.indexOf(name)
        if (index < 0) return
        if (binding.viewPager.currentItem != index) {
            binding.viewPager.setCurrentItem(index, true)
        }
    }

    private fun scrollToTopAt(index: Int) {
        val tag = "f$index"
        val frag = childFragmentManager.findFragmentByTag(tag)
        if (frag is VideoListFragment) {
            frag.scrollToTop()
        }
    }

    private fun applyTabStates(progress: Float) {
        for (i in tabs.indices) {
            val dist = kotlin.math.abs(progress - i)
            val weight = (1f - dist).coerceIn(0f, 1f)

            val tabView = when (tabs[i]) {
                "recommended" -> binding.tabRecommended
                "latest" -> binding.tabLatest
                "downloads" -> binding.tabDownloads
                else -> null
            }
            tabView?.let { applyTabStateToView(it, weight) }
        }
    }

    private fun applyTabStateToView(tab: View, weight: Float) {
        val scale = 0.94f + 0.06f * weight
        tab.scaleX = scale
        tab.scaleY = scale
        tab.alpha = 0.6f + 0.4f * weight
        tab.background?.mutate()?.alpha = (weight * 255).toInt().coerceIn(0, 255)
    }

    override fun onResume() {
        super.onResume()
        // На всякий случай — проверяем и здесь (если что-то поменялось через другие пути)
        refreshTabsIfNeeded()
    }

    override fun onDestroyView() {
        super.onDestroyView()

        // Отписываемся от prefs
        if (prefsListener != null) {
            val prefs = requireContext()
                .getSharedPreferences("asmr1_downloads", Context.MODE_PRIVATE)
            prefs.unregisterOnSharedPreferenceChangeListener(prefsListener)
            prefsListener = null
        }

        _binding = null
    }
}

package ru.asmr1.app.ui

import android.os.Bundle
import android.text.Editable
import android.text.TextWatcher
import android.view.LayoutInflater
import android.view.View
import android.view.ViewGroup
import android.view.inputmethod.EditorInfo
import android.view.inputmethod.InputMethodManager
import android.widget.TextView
import android.widget.Toast
import androidx.core.content.ContextCompat
import androidx.fragment.app.Fragment
import androidx.lifecycle.lifecycleScope
import androidx.recyclerview.widget.LinearLayoutManager
import androidx.recyclerview.widget.RecyclerView
import kotlinx.coroutines.Job
import kotlinx.coroutines.delay
import kotlinx.coroutines.launch
import ru.asmr1.app.MainActivity

import ru.asmr1.app.utils.SearchHistoryManager

class SearchFragment : Fragment() {

    private var _binding: FragmentSearchBinding? = null
    private val binding get() = _binding!!

    private lateinit var adapter: SearchAdapter
    private lateinit var historyAdapter: SearchHistoryAdapter
    private lateinit var layoutManager: LinearLayoutManager

    private lateinit var historyManager: SearchHistoryManager

    private var currentQuery = ""
    private var currentPage = 1
    private var isLoading = false
    private var hasMore = true
    private val perPage = 20

    // Сортировка
    private var orderBy: String = "weight"
    private var order: String = "DESC"

    private var searchJob: Job? = null
    private var searchDebounceJob: Job? = null

    override fun onCreateView(
        inflater: LayoutInflater,
        container: ViewGroup?,
        savedInstanceState: Bundle?
    ): View {
        _binding = FragmentSearchBinding.inflate(inflater, container, false)
        return binding.root
    }

    override fun onViewCreated(view: View, savedInstanceState: Bundle?) {
        super.onViewCreated(view, savedInstanceState)

        historyManager = SearchHistoryManager(requireContext())

        setupRecyclerView()
        setupHistoryRecycler()
        setupSearchInput()
        setupButtons()
        setupSortChips()

        showInitialState()

        binding.etSearch.postDelayed({
            if (_binding == null) return@postDelayed
            binding.etSearch.requestFocus()
            val imm = ContextCompat.getSystemService(requireContext(), InputMethodManager::class.java)
            imm?.showSoftInput(binding.etSearch, InputMethodManager.SHOW_IMPLICIT)
        }, 200)
    }

    private fun setupRecyclerView() {
        adapter = SearchAdapter { video ->
            (requireActivity() as? MainActivity)?.openPlayer(video)
        }
        layoutManager = LinearLayoutManager(requireContext())
        binding.recyclerView.layoutManager = layoutManager
        binding.recyclerView.adapter = adapter
        binding.recyclerView.setHasFixedSize(false)
        binding.recyclerView.itemAnimator = null

        binding.recyclerView.addOnScrollListener(object : RecyclerView.OnScrollListener() {
            override fun onScrolled(recyclerView: RecyclerView, dx: Int, dy: Int) {
                super.onScrolled(recyclerView, dx, dy)
                if (dy <= 0) return
                if (isLoading || !hasMore) return
                if (_binding == null) return

                val totalItems = layoutManager.itemCount
                val lastVisible = layoutManager.findLastVisibleItemPosition()
                if (totalItems > 0 && lastVisible >= totalItems - 5) {
                    currentPage++
                    performSearch(currentQuery, currentPage, append = true)
                }
            }
        })

        binding.swipeRefresh.setOnRefreshListener {
            if (currentQuery.isNotEmpty()) {
                currentPage = 1
                hasMore = true
                performSearch(currentQuery, 1, append = false)
            } else {
                binding.swipeRefresh.isRefreshing = false
            }
        }
    }

    private fun setupHistoryRecycler() {
        historyAdapter = SearchHistoryAdapter(
            items = emptyList(),
            onItemClick = { query ->
                binding.etSearch.setText(query)
                binding.etSearch.setSelection(query.length)
                currentQuery = query
                currentPage = 1
                hasMore = true
                performSearch(query, 1, append = false)
                hideKeyboard()
            },
            onItemDelete = { query ->
                historyManager.removeQuery(query)
                refreshHistory()
            }
        )
        binding.rvHistory.layoutManager = LinearLayoutManager(requireContext())
        binding.rvHistory.adapter = historyAdapter
    }

    private fun setupSearchInput() {
        binding.etSearch.addTextChangedListener(object : TextWatcher {
            override fun beforeTextChanged(s: CharSequence?, start: Int, count: Int, after: Int) {}
            override fun onTextChanged(s: CharSequence?, start: Int, before: Int, count: Int) {
                val hasText = !s.isNullOrEmpty()
                binding.btnClear.visibility = if (hasText) View.VISIBLE else View.GONE

                val query = s?.toString()?.trim().orEmpty()

                if (query.isEmpty()) {
                    showInitialState()
                    return
                }

                searchDebounceJob?.cancel()
                searchDebounceJob = viewLifecycleOwner.lifecycleScope.launch {
                    delay(500)
                    currentQuery = query
                    currentPage = 1
                    hasMore = true
                    performSearch(query, 1, append = false)
                }
            }
            override fun afterTextChanged(s: Editable?) {}
        })

        binding.etSearch.setOnFocusChangeListener { _, hasFocus ->
            if (hasFocus && binding.etSearch.text.isNullOrEmpty()) {
                showInitialState()
            }
        }

        binding.etSearch.setOnEditorActionListener { _, actionId, _ ->
            if (actionId == EditorInfo.IME_ACTION_SEARCH) {
                searchDebounceJob?.cancel()
                val query = binding.etSearch.text?.toString()?.trim().orEmpty()
                if (query.isNotEmpty()) {
                    historyManager.addQuery(query)
                    currentQuery = query
                    currentPage = 1
                    hasMore = true
                    performSearch(query, 1, append = false)
                    hideKeyboard()
                }
                true
            } else false
        }
    }

    private fun setupButtons() {
        binding.btnSearch.setOnClickListener {
            searchDebounceJob?.cancel()
            val query = binding.etSearch.text?.toString()?.trim().orEmpty()
            if (query.isNotEmpty()) {
                historyManager.addQuery(query)
                currentQuery = query
                currentPage = 1
                hasMore = true
                performSearch(query, 1, append = false)
                hideKeyboard()
            } else {
                Toast.makeText(context, "Введите запрос", Toast.LENGTH_SHORT).show()
            }
        }

        binding.btnClear.setOnClickListener {
            binding.etSearch.text?.clear()
            adapter.clear()
            showInitialState()
        }

        binding.btnClearHistory.setOnClickListener {
            historyManager.clearAll()
            showInitialState()
        }
    }

    // ===== ЧИПЫ СОРТИРОВКИ =====
    private fun setupSortChips() {
        binding.chipWeight.setOnClickListener { selectOrderBy("weight") }
        binding.chipDate.setOnClickListener { selectOrderBy("date") }
        binding.chipViews.setOnClickListener { selectOrderBy("views") }
        binding.chipLikes.setOnClickListener { selectOrderBy("likes") }
        binding.chipTitle.setOnClickListener { selectOrderBy("title") }

        binding.chipOrder.setOnClickListener {
            order = if (order == "DESC") "ASC" else "DESC"
            updateOrderChipText()
            applySortAndSearch()
        }

        updateSortChipsUI()
        updateOrderChipText()
    }

    private fun selectOrderBy(newOrderBy: String) {
        if (orderBy == newOrderBy) return
        orderBy = newOrderBy
        updateSortChipsUI()
        applySortAndSearch()
    }

    private fun updateSortChipsUI() {
        _binding?.apply {
            setChipActive(chipWeight, orderBy == "weight")
            setChipActive(chipDate, orderBy == "date")
            setChipActive(chipViews, orderBy == "views")
            setChipActive(chipLikes, orderBy == "likes")
            setChipActive(chipTitle, orderBy == "title")
        }
    }

    private fun setChipActive(chip: TextView, active: Boolean) {
        chip.isSelected = active
        chip.setTextColor(
            ContextCompat.getColor(
                requireContext(),
                if (active) android.R.color.white else R.color.text_secondary
            )
        )
    }

    private fun updateOrderChipText() {
        _binding?.chipOrder?.text = if (order == "DESC") "↓ Убыв." else "↑ Возр."
    }

    /** Если есть активный запрос — перезапускаем поиск с новой сортировкой */
    private fun applySortAndSearch() {
        val query = binding.etSearch.text?.toString()?.trim().orEmpty()
        if (query.isNotEmpty()) {
            currentQuery = query
            currentPage = 1
            hasMore = true
            performSearch(query, 1, append = false)
        }
    }

    private fun performSearch(query: String, page: Int, append: Boolean) {
        if (isLoading) return
        if (query.isEmpty()) return

        isLoading = true

        if (!append) {
            showSkeletons()
            _binding?.apply {
                llHistory.visibility = View.GONE
                llEmpty.visibility = View.GONE
                llNoResults.visibility = View.GONE
                swipeRefresh.visibility = View.VISIBLE
                llSortBar.visibility = View.VISIBLE
            }
        }

        searchJob?.cancel()
        searchJob = viewLifecycleOwner.lifecycleScope.launch {
            try {
                val response = ApiClient.service.search(
                    query = query,
                    page = page,
                    perPage = perPage,
                    orderBy = orderBy,
                    order = order
                )

                if (_binding == null) return@launch

                if (response.success) {
                    if (append) {
                        adapter.appendVideos(response.videos)
                    } else {
                        adapter.setVideos(response.videos)
                    }
                    hasMore = response.hasMore

                    if (response.videos.isEmpty()) {
                        showNoResults(query)
                    } else {
                        showResults()
                    }
                } else {
                    if (!append) showNoResults(query)
                }
            } catch (e: Exception) {
                if (_binding == null) return@launch
                Toast.makeText(context, "Ошибка поиска: ${e.message}", Toast.LENGTH_SHORT).show()
                if (!append) showNoResults(query)
            } finally {
                isLoading = false
                _binding?.swipeRefresh?.isRefreshing = false
            }
        }
    }

    private fun showInitialState() {
        _binding?.apply {
            swipeRefresh.visibility = View.GONE

class SearchAdapter(
    private var videos: List<Video> = emptyList(),
    private val onClick: (Video) -> Unit = {}

    companion object {
        private const val TYPE_VIDEO = 0
        private const val TYPE_SKELETON = 1
        private const val SKELETON_COUNT = 6

        // Общий SimpleDateFormat (с UTC)
        private val dateFmt = SimpleDateFormat("yyyy-MM-dd HH:mm:ss", Locale.US).apply {
            timeZone = TimeZone.getTimeZone("UTC")
        }
    }

    private var showSkeleton = false

    fun setVideos(newVideos: List<Video>) {
        val wasSkeleton = showSkeleton
        showSkeleton = false

        if (wasSkeleton) {
            videos = newVideos
            notifyDataSetChanged()
            return
        }

        val old = videos
        val diff = DiffUtil.calculateDiff(object : DiffUtil.Callback() {
            override fun getOldListSize() = old.size
            override fun getNewListSize() = newVideos.size
            override fun areItemsTheSame(o: Int, n: Int) = old[o].id == newVideos[n].id
            override fun areContentsTheSame(o: Int, n: Int) = old[o] == newVideos[n]
        })
        videos = newVideos
        diff.dispatchUpdatesTo(this)
    }

    fun appendVideos(newVideos: List<Video>) {
        if (newVideos.isEmpty()) return
        val startPos = videos.size
        videos = videos + newVideos
        notifyItemRangeInserted(startPos, newVideos.size)
    }

    fun showSkeletons() {
        showSkeleton = true
        videos = emptyList()
        notifyDataSetChanged()
    }

    fun clear() {
        showSkeleton = false
        videos = emptyList()
        notifyDataSetChanged()
    }

    override fun getItemViewType(position: Int): Int =
        if (showSkeleton) TYPE_SKELETON else TYPE_VIDEO

    override fun getItemCount(): Int =
        if (showSkeleton) SKELETON_COUNT else videos.size

    override fun onCreateViewHolder(parent: ViewGroup, viewType: Int): RecyclerView.ViewHolder {
        val inflater = LayoutInflater.from(parent.context)
        return when (viewType) {
            TYPE_SKELETON -> SkeletonViewHolder(
                ItemVideoSkeletonSearchBinding.inflate(inflater, parent, false)
            )
            else -> VideoViewHolder(
                ItemVideoSearchBinding.inflate(inflater, parent, false)
            )
        }
    }

    override fun onBindViewHolder(holder: RecyclerView.ViewHolder, position: Int) {
        when (holder) {
            is VideoViewHolder -> holder.bind(videos[position])
            is SkeletonViewHolder -> holder.bind()
        }
    }

    private fun formatViewsShort(n: Int): String = when {
        n >= 1_000_000 -> String.format(Locale.US, "%.1fM", n / 1_000_000.0)
        n >= 1_000 -> String.format(Locale.US, "%.1fK", n / 1_000.0)
        else -> n.toString()
    }

    private fun timeAgoShort(dateStr: String): String {
        if (dateStr.isEmpty()) return ""
        return try {
            val date = synchronized(dateFmt) { dateFmt.parse(dateStr) } ?: return ""
            val diff = (System.currentTimeMillis() - date.time) / 1000
            when {
                diff < 60 -> "сейчас"
                diff < 3600 -> "${diff / 60} мин"
                diff < 86400 -> "${diff / 3600} ч"
                diff < 2_592_000 -> "${diff / 86400} дн"
                diff < 31_536_000 -> "${diff / 2_592_000} мес"
                else -> "${diff / 31_536_000} г"
            }
        } catch (_: Exception) {
            ""
        }
    }

                    error(R.color.card_bg)

                    allowHardware(true)
                }

                binding.duration.visibility = View.VISIBLE
                binding.duration.text = formatDuration(video.duration)

            }

        }

    private fun setupAuthorsRecycler() {

                selectedAuthorId = 0

                authorsAdapter.setSelected(author.id)
                loadVideos(reset = true)
            }
        )
        binding.rvAuthors.layoutManager = LinearLayoutManager(
            requireContext(), LinearLayoutManager.HORIZONTAL, false
        )
        binding.rvAuthors.adapter = authorsAdapter
    }

    private fun setupVideosRecycler() {
        videosAdapter = VideoAdapter(
            onClick = { video ->
                markAuthorAsViewed(video.author.id)
                (requireActivity() as? MainActivity)?.openPlayer(video)
            },
            onMoreClick = { _, _ -> }
        )
        videosLayoutManager = LinearLayoutManager(requireContext())
        binding.rvVideos.layoutManager = videosLayoutManager
        binding.rvVideos.adapter = videosAdapter
        binding.rvVideos.setHasFixedSize(false)
        binding.rvVideos.itemAnimator = null

        binding.rvVideos.addOnScrollListener(object : RecyclerView.OnScrollListener() {
            override fun onScrolled(recyclerView: RecyclerView, dx: Int, dy: Int) {
                super.onScrolled(recyclerView, dx, dy)
                if (dy <= 0) return
                if (isLoading || !hasMore) return
                if (_binding == null) return

                val total = videosLayoutManager.itemCount
                val lastVisible = videosLayoutManager.findLastVisibleItemPosition()
                if (total > 0 && lastVisible >= total - 5) {
                    currentPage++
                    loadVideos(reset = false)
                }
            }
        })
    }

    private fun setupSwipeRefresh() {
        binding.swipeRefresh.setOnRefreshListener {
            loadAuthors()
            loadVideos(reset = true)
        }
    }

    private fun loadAuthors() {
        viewLifecycleOwner.lifecycleScope.launch {
            try {
                val token = authManager.getToken() ?: return@launch
                val api = ApiClient.createJwtService(token)
                val response = api.getSubscribedAuthors()

                if (_binding == null) return@launch

                if (response.success) {
                    authorsAdapter.setAuthors(response.authors)
                    if (response.authors.isNotEmpty()) {
                        binding.rvAuthors.visibility = View.VISIBLE
                        binding.llEmpty.visibility = View.GONE
                    } else {
                        binding.rvAuthors.visibility = View.GONE

        }
    }

        if (reset) {
            currentPage = 1
            hasMore = true
        }

        loadJob?.cancel()
        loadJob = viewLifecycleOwner.lifecycleScope.launch {

                    page = currentPage,

                        binding.swipeRefresh.visibility = View.GONE
                    } else {
                        binding.swipeRefresh.visibility = View.VISIBLE
                        binding.llEmpty.visibility = View.GONE
                    }
                }
            } catch (e: Exception) {
                if (_binding == null) return@launch
                Toast.makeText(context, "Ошибка: ${e.message}", Toast.LENGTH_SHORT).show()
            } finally {
                isLoading = false
                _binding?.swipeRefresh?.isRefreshing = false
            }
        }
    }

            try {
                val token = authManager.getToken() ?: return@launch

                api.markAuthorViewed(authorId)
            } catch (_: Exception) { }
        }
    }

            llEmpty.visibility = View.VISIBLE
        }
    }

        loadJob?.cancel()
        _binding = null
    }
}

            binding.authorName.text = item.authorName

            // Тип: видео или аудио
            binding.typeIcon.setImageResource(
                if (item.type == DownloadedFile.Type.VIDEO) R.drawable.ic_download_video
                else R.drawable.ic_download_audio
            )

            // Размер
            binding.sizeText.text = formatSize(item.size)

            // Дата скачивания
            binding.downloadDate.text = "Скачано ${timeAgo(item.downloadedAt)}"

            // Превью
            if (item.thumbnail.isNotEmpty()) {
                binding.thumbnail.load(item.thumbnail) {
                    crossfade(false)
                    placeholder(R.color.card_bg)
                    error(R.color.card_bg)
                    memoryCachePolicy(CachePolicy.ENABLED)
                    diskCachePolicy(CachePolicy.ENABLED)
                    allowHardware(true)
                }
            }

            // Аватар автора
            if (item.authorAvatar.isNotEmpty()) {
                binding.authorAvatar.load(item.authorAvatar) {
                    crossfade(false)
                    placeholder(R.color.card_bg)
                    error(R.color.card_bg)
                    memoryCachePolicy(CachePolicy.ENABLED)
                    diskCachePolicy(CachePolicy.ENABLED)
                }
            }

            binding.root.setOnClickListener { onClick(item) }
            binding.btnDelete.setOnClickListener { onDelete(item) }
        }
    }

    private fun formatSize(bytes: Long): String {
        if (bytes <= 0) return "0 МБ"
        val kb = bytes / 1024.0
        val mb = kb / 1024.0
        val gb = mb / 1024.0
        return when {
            gb >= 1.0 -> String.format(Locale.US, "%.1f ГБ", gb)
            mb >= 1.0 -> String.format(Locale.US, "%.0f МБ", mb)
            else -> String.format(Locale.US, "%.0f КБ", kb)
        }
    }

    private fun timeAgo(timestamp: Long): String {
        val diff = System.currentTimeMillis() - timestamp
        val minutes = TimeUnit.MILLISECONDS.toMinutes(diff)
        val hours = TimeUnit.MILLISECONDS.toHours(diff)
        val days = TimeUnit.MILLISECONDS.toDays(diff)
        return when {
            minutes < 1 -> "только что"
            minutes < 60 -> "$minutes мин назад"
            hours < 24 -> "$hours ч назад"
            days < 30 -> "$days дн назад"
            else -> "${days / 30} мес назад"
        }
    }
}

package ru.asmr1.app.auth

import android.content.Context
import androidx.security.crypto.EncryptedSharedPreferences
import androidx.security.crypto.MasterKey

class AuthManager(context: Context) {

    private val masterKey = MasterKey.Builder(context)
        .setKeyScheme(MasterKey.KeyScheme.AES256_GCM)
        .build()

    private val prefs = EncryptedSharedPreferences.create(
        context,
        "asmr1_auth_prefs",
        masterKey,
        EncryptedSharedPreferences.PrefKeyEncryptionScheme.AES256_SIV,
        EncryptedSharedPreferences.PrefValueEncryptionScheme.AES256_GCM
    )

    fun saveAuth(
        token: String,
        username: String,
        displayName: String,
        email: String,
        userId: Int
    ) {
        prefs.edit()
            .putString(KEY_TOKEN, token)
            .putString(KEY_USERNAME, username)
            .putString(KEY_DISPLAY_NAME, displayName)
            .putString(KEY_EMAIL, email)
            .putInt(KEY_USER_ID, userId)
            .putLong(KEY_LOGIN_TIME, System.currentTimeMillis())
            .apply()
    }

    fun getToken(): String? = prefs.getString(KEY_TOKEN, null)
    fun getUsername(): String? = prefs.getString(KEY_USERNAME, null)
    fun getDisplayName(): String? = prefs.getString(KEY_DISPLAY_NAME, null)
    fun getEmail(): String? = prefs.getString(KEY_EMAIL, null)
    fun getUserId(): Int = prefs.getInt(KEY_USER_ID, 0)
    fun getLoginTime(): Long = prefs.getLong(KEY_LOGIN_TIME, 0L)

    fun isLoggedIn(): Boolean = !getToken().isNullOrEmpty()

    fun clear() {
        prefs.edit().clear().apply()
    }

    companion object {
        private const val KEY_TOKEN = "jwt_token"
        private const val KEY_USERNAME = "jwt_username"
        private const val KEY_DISPLAY_NAME = "jwt_display_name"
        private const val KEY_EMAIL = "jwt_email"
        private const val KEY_USER_ID = "jwt_user_id"
        private const val KEY_LOGIN_TIME = "jwt_login_time"
    }
}

import okhttp3.Interceptor
import okhttp3.Response

class AuthInterceptor(private val token: String) : Interceptor {
    override fun intercept(chain: Interceptor.Chain): Response {
        val request = chain.request().newBuilder()
            .addHeader("Authorization", "Bearer $token")
            .build()
        return chain.proceed(request)
    }
}

            toast("Заполните все поля")
            return

        viewLifecycleOwner.lifecycleScope.launch {
            try {
                val response = ApiClient.authApi.login(
                    LoginRequest(username = username, password = password)
                )
                val token = response.token

                authManager.saveAuth(
                    token = token,
                    username = response.userNicename,
                    displayName = response.userDisplayName,
                    email = response.userEmail,
                    userId = 0
                )

                try {
                    val userApi = ApiClient.createJwtAuthService(token)
                    val user = userApi.getCurrentUser()
                    authManager.saveAuth(
                        token = token,
                        username = user.slug ?: response.userNicename,
                        displayName = user.name,
                        email = response.userEmail,
                        userId = user.id
                    )
                } catch (_: Exception) { }

                toast("Добро пожаловать, ${response.userDisplayName}!")
                (activity as? MainActivity)?.onLoginSuccess()

            } catch (e: Exception) {
                showLoading(false)
                val msg = e.message ?: "Неизвестная ошибка"
                val cleanMsg = when {
                    msg.contains("401") || msg.contains("403") -> "Неверный логин или пароль"
                    msg.contains("timeout", true) -> "Сервер не отвечает"
                    else -> "Ошибка: $msg"
                }
                toast(cleanMsg)
            }
        }
    }

    private fun showLoading(show: Boolean) {
        _binding?.apply {
            progressBar.visibility = if (show) View.VISIBLE else View.GONE
            btnLogin.isEnabled = !show
            btnRegister.isEnabled = !show
            btnGuest.isEnabled = !show
            etUsername.isEnabled = !show
            etPassword.isEnabled = !show
        }

        parentFragmentManager.popBackStack()

    }

        _binding = null
    }
}

            try {

                if (_binding == null) return@launch

                    captchaId = response.captchaId
                    binding.tvCaptchaQuestion.text = "${response.question} ="
                    captchaLoaded = true
                } else {
                    binding.tvCaptchaQuestion.text = "Ошибка капчи"
                    toast("Не удалось загрузить капчу: ${response.message ?: "?"}")
                }
            } catch (e: Exception) {

        }
        if (!captchaLoaded || captchaId.isEmpty()) {

            loadCaptcha()
            return
        }

        showLoading(true)
        isSubmitting = true

        viewLifecycleOwner.lifecycleScope.launch {
            try {
                val response = ApiClient.authApi.register(
                    RegisterRequest(
                        username = username,
                        email = email,

                        captchaAnswer = captchaAnswer
                    )

                if (response.success && !response.token.isNullOrEmpty()) {

                        username = response.userNicename ?: username,
                        displayName = response.userDisplayName ?: username,
                        email = response.userEmail ?: email,
                        userId = response.userId ?: 0
                    )

                    // Обновляем профиль (как в LoginFragment)
                    try {
                        val userApi = ApiClient.createJwtAuthService(response.token)
                        val user = userApi.getCurrentUser()
                        authManager.saveAuth(
                            token = response.token,
                            username = user.slug ?: (response.userNicename ?: username),
                            displayName = user.name,
                            email = response.userEmail ?: email,
                            userId = user.id
                        )
                    } catch (_: Exception) { }

                    toast("Добро пожаловать, ${response.userDisplayName ?: username}!")
                    (activity as? MainActivity)?.onLoginSuccess()

                } else {
                    // Ошибка от сервера
                    showLoading(false)
                    isSubmitting = false

                    val errMsg = response.message ?: "Не удалось зарегистрироваться"
                    toast(errMsg)

                    // Если проблема с капчей — грузим новую
                    val errCode = response.code ?: ""
                    if (errCode.startsWith("captcha")) {
                        loadCaptcha()
                    }
                }

            } catch (e: Exception) {
                if (_binding == null) return@launch
                showLoading(false)
                isSubmitting = false

                val cleanMsg = when {

                    else -> "Ошибка: $msg"
                }
                toast(cleanMsg)
                loadCaptcha()
            }
        }
    }

fun formatViews(n: Int): String = when {
    n >= 1_000_000 -> String.format("%.1fM просмотров", n / 1_000_000.0)
    n >= 1_000 -> String.format("%.1fK просмотров", n / 1_000.0)
    else -> "$n просмотров"
}

fun formatDuration(sec: Int): String {
    val m = sec / 60
    val s = sec % 60
    return "$m:${if (s < 10) "0$s" else s}"
}

fun timeAgo(dateStr: String): String {
    if (dateStr.isEmpty()) return ""
    return try {
        val fmt = SimpleDateFormat("yyyy-MM-dd HH:mm:ss", Locale.US).apply {
            timeZone = TimeZone.getTimeZone("UTC")
        }
        val date = fmt.parse(dateStr) ?: return ""
        val diff = (System.currentTimeMillis() - date.time) / 1000
        when {
            diff < 60 -> "только что"
            diff < 3600 -> "${diff / 60} мин назад"
            diff < 86400 -> "${diff / 3600} ч назад"
            diff < 2592000 -> "${diff / 86400} дн назад"
            diff < 31536000 -> "${diff / 2592000} мес назад"
            else -> "${diff / 31536000} г назад"
        }
    } catch (_: Exception) {
        ""
    }
}

package ru.asmr1.app.utils

import android.content.Context

/**
 * Локальная история поиска — сохраняется в SharedPreferences.
 * Максимум 20 запросов, новые сверху, без дубликатов.
 */
class SearchHistoryManager(context: Context) {

    private val prefs = context.getSharedPreferences("asmr1_search_history", Context.MODE_PRIVATE)

    companion object {
        private const val KEY_HISTORY = "history_items"
        private const val SEPARATOR = "|||"
        private const val MAX_ITEMS = 20
    }

    fun getHistory(): List<String> {
        val raw = prefs.getString(KEY_HISTORY, null) ?: return emptyList()
        if (raw.isEmpty()) return emptyList()
        return raw.split(SEPARATOR).filter { it.isNotBlank() }
    }

    fun addQuery(query: String) {
        val trimmed = query.trim()
        if (trimmed.isEmpty()) return

        val current = getHistory().toMutableList()
        current.removeAll { it.equals(trimmed, ignoreCase = true) }
        current.add(0, trimmed)
        val limited = current.take(MAX_ITEMS)
        prefs.edit().putString(KEY_HISTORY, limited.joinToString(SEPARATOR)).apply()
    }

    fun removeQuery(query: String) {
        val current = getHistory().toMutableList()
        current.removeAll { it == query }
        prefs.edit().putString(KEY_HISTORY, current.joinToString(SEPARATOR)).apply()
    }

    fun clearAll() {
        prefs.edit().remove(KEY_HISTORY).apply()
    }

    fun isEmpty(): Boolean = getHistory().isEmpty()
}

package ru.asmr1.app.download

import android.content.Context
import com.google.gson.Gson
import com.google.gson.reflect.TypeToken
import ru.asmr1.app.data.Video
import java.io.File

/**
 * Хранилище метаданных о скачанных файлах.
 * Файлы — context.filesDir/downloads/
 * Метаданные — SharedPreferences (JSON).
 */
class DownloadStorage(context: Context) {

    private val prefs = context.getSharedPreferences("asmr1_downloads", Context.MODE_PRIVATE)
    private val gson = Gson()
    private val appContext = context.applicationContext

    companion object {
        private const val KEY_INDEX = "downloads_index"

        fun getDownloadsDir(context: Context): File {
            val dir = File(context.filesDir, "downloads")
            if (!dir.exists()) dir.mkdirs()
            return dir
        }
    }

    private fun readIndex(): MutableMap<String, DownloadedFile> {
        val json = prefs.getString(KEY_INDEX, null) ?: return mutableMapOf()
        return try {
            val type = object : TypeToken<MutableMap<String, DownloadedFile>>() {}.type
            gson.fromJson(json, type) ?: mutableMapOf()
        } catch (e: Exception) {
            mutableMapOf()
        }
    }

    private fun writeIndex(index: Map<String, DownloadedFile>) {
        prefs.edit().putString(KEY_INDEX, gson.toJson(index)).apply()
    }

    private fun key(videoId: Int, type: DownloadedFile.Type): String =
        "${videoId}_${type.name.lowercase()}"

    // ===== Чтение =====

    fun getFile(videoId: Int, type: DownloadedFile.Type): DownloadedFile? {
        val index = readIndex()
        val entry = index[key(videoId, type)] ?: return null
        val f = File(getDownloadsDir(appContext), entry.fileName)
        return if (f.exists()) entry else {
            index.remove(key(videoId, type))
            writeIndex(index)
            null
        }
    }

    fun hasVideo(videoId: Int): Boolean = getFile(videoId, DownloadedFile.Type.VIDEO) != null
    fun hasAudio(videoId: Int): Boolean = getFile(videoId, DownloadedFile.Type.AUDIO) != null

    fun hasAnyDownloads(): Boolean = readIndex().isNotEmpty()

    /**
     * Все скачанные файлы (для экрана «Скачано»).
     * По умолчанию — сортировка по дате скачивания (свежие первыми).
     */
    fun getAllDownloads(): List<DownloadedFile> {
        val all = readIndex().values.filter { entry ->
            val f = File(getDownloadsDir(appContext), entry.fileName)
            f.exists()
        }
        return all.sortedByDescending { it.downloadedAt }
    }

    /**
     * Скачанные файлы конкретного типа.
     * null = все типы.
     */
    fun getDownloads(type: DownloadedFile.Type? = null): List<DownloadedFile> {
        val all = getAllDownloads()
        return if (type == null) all else all.filter { it.type == type }
    }

    // ===== Запись =====

    fun saveFile(video: Video, type: DownloadedFile.Type, file: File) {
        val index = readIndex()
        index[key(video.id, type)] = DownloadedFile(
            videoId = video.id,
            type = type,
            fileName = file.name,
            size = file.length(),
            downloadedAt = System.currentTimeMillis(),
            title = video.title,
            description = video.description,
            thumbnail = video.thumbnail,
            duration = video.duration,
            views = video.views,
            likes = video.likes,
            published = video.published,
            videoUrl = video.videoUrl,
            audioUrl = video.audioUrl,
            authorId = video.author.id,
            authorName = video.author.name,
            authorAvatar = video.author.avatar
        )
        writeIndex(index)
    }

    fun deleteFile(videoId: Int, type: DownloadedFile.Type): Boolean {
        val entry = getFile(videoId, type) ?: return false
        val file = File(getDownloadsDir(appContext), entry.fileName)
        val deleted = if (file.exists()) file.delete() else true
        if (deleted) {
            val index = readIndex()
            index.remove(key(videoId, type))
            writeIndex(index)
        }
        return deleted
    }

    fun getFilePath(videoId: Int, type: DownloadedFile.Type): File? {
        val entry = getFile(videoId, type) ?: return null
        return File(getDownloadsDir(appContext), entry.fileName)
    }
}

package ru.asmr1.app.download

import android.content.Context
import kotlinx.coroutines.Dispatchers
import kotlinx.coroutines.flow.Flow
import kotlinx.coroutines.flow.flow
import kotlinx.coroutines.flow.flowOn
import okhttp3.OkHttpClient
import okhttp3.Request
import ru.asmr1.app.data.Video
import java.io.File
import java.util.concurrent.TimeUnit

/**
 * Движок загрузки. Качает во внутреннее хранилище приложения.
 */
class Downloader(private val context: Context) {

    private val client = OkHttpClient.Builder()
        .connectTimeout(30, TimeUnit.SECONDS)
        .readTimeout(60, TimeUnit.SECONDS)
        .build()

    private val storage = DownloadStorage(context)

    /**
     * Скачать файл. При завершении сохраняет полные метаданные видео.
     */
    fun download(
        video: Video,
        type: DownloadedFile.Type,
        url: String,
        extension: String
    ): Flow<DownloadProgress> = flow {

        val existing = storage.getFile(video.id, type)
        if (existing != null) {
            emit(DownloadProgress(video.id, type, existing.size, existing.size, 100))
            return@flow
        }

        val dir = DownloadStorage.getDownloadsDir(context)
        val finalName = "${type.name.lowercase()}_${video.id}.${extension}"
        val tmpName = "${finalName}.part"
        val tmpFile = File(dir, tmpName)
        val finalFile = File(dir, finalName)

        if (tmpFile.exists()) tmpFile.delete()

        val request = Request.Builder()
            .url(url)
            .header("User-Agent", "ASMR1-Android/1.0")
            .build()

        client.newCall(request).execute().use { response ->
            if (!response.isSuccessful) throw RuntimeException("HTTP ${response.code}")
            val body = response.body ?: throw RuntimeException("Empty body")

            val total = body.contentLength()
            var downloaded = 0L

            body.byteStream().use { input ->
                tmpFile.outputStream().use { output ->
                    val buffer = ByteArray(8 * 1024)
                    var lastEmit = 0L
                    while (true) {
                        val read = input.read(buffer)
                        if (read <= 0) break
                        output.write(buffer, 0, read)
                        downloaded += read

                        val now = System.currentTimeMillis()
                        if (now - lastEmit >= 200 || (total > 0 && downloaded >= total)) {
                            lastEmit = now
                            val percent = if (total > 0) {
                                ((downloaded.toDouble() / total) * 100).toInt().coerceIn(0, 100)
                            } else 0
                            emit(DownloadProgress(video.id, type, downloaded, if (total > 0) total else 0, percent))
                        }
                    }
                }
            }
        }

        if (finalFile.exists()) finalFile.delete()
        val ok = tmpFile.renameTo(finalFile)
        if (!ok) {
            tmpFile.copyTo(finalFile, overwrite = true)
            tmpFile.delete()
        }

        storage.saveFile(video, type, finalFile)

        emit(DownloadProgress(video.id, type, finalFile.length(), finalFile.length(), 100))
    }.flowOn(Dispatchers.IO)

    fun delete(videoId: Int, type: DownloadedFile.Type): Boolean {
        return storage.deleteFile(videoId, type)
    }

    fun getRemoteFileSize(url: String): Long {
        return try {
            val request = Request.Builder()
                .url(url)
                .head()
                .header("User-Agent", "ASMR1-Android/1.0")
                .build()
            client.newCall(request).execute().use { response ->
                if (!response.isSuccessful) return -1
                response.header("Content-Length")?.toLongOrNull() ?: -1
            }
        } catch (e: Exception) {
            -1
        }
    }
}

package ru.asmr1.app.download

import android.content.Context
import android.net.Uri
import ru.asmr1.app.data.Video
import java.io.File

/**
 * Определяет, откуда играть видео:
 * - Скачано видео → локальный .mp4
 * - Скачано аудио → локальный .mp3 (в аудио-режиме)
 * - Ничего не скачано → онлайн URL
 */
object PlaybackModeResolver {

    sealed class Mode {
        /** Играем видео (локально или онлайн) */
        data class VideoMode(val uri: Uri) : Mode()

        /** Играем только аудио — вместо видео показываем превью */
        data class AudioMode(
            val uri: Uri,
            val previewUrl: String
        ) : Mode()
    }

    /**
     * @param video объект видео
     * @return Mode — как играть
     */
    fun resolve(context: Context, video: Video): Mode {
        val storage = DownloadStorage(context)

        val hasVideo = storage.hasVideo(video.id)
        val hasAudio = storage.hasAudio(video.id)

        // Приоритет: локальное видео > локальное аудио > онлайн видео
        return when {
            hasVideo -> {
                val file = storage.getFilePath(video.id, DownloadedFile.Type.VIDEO)
                if (file != null && file.exists()) {
                    Mode.VideoMode(Uri.fromFile(file))
                } else {
                    // Файл пропал — играем онлайн
                    Mode.VideoMode(Uri.parse(video.videoUrl))
                }
            }
            hasAudio -> {
                val file = storage.getFilePath(video.id, DownloadedFile.Type.AUDIO)
                if (file != null && file.exists()) {
                    Mode.AudioMode(
                        uri = Uri.fromFile(file),
                        previewUrl = video.thumbnail
                    )
                } else {
                    Mode.VideoMode(Uri.parse(video.videoUrl))
                }
            }
            else -> {
                // Ничего не скачано — онлайн
                Mode.VideoMode(Uri.parse(video.videoUrl))
            }
        }
    }

    /**
     * Просто определить, играем ли мы локальный файл (полезно для UI).
     */
    fun isOffline(context: Context, videoId: Int): Boolean {
        val storage = DownloadStorage(context)
        return storage.hasVideo(videoId) || storage.hasAudio(videoId)
    }
}
