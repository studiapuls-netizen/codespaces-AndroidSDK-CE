    private fun setupSplash() {
        val splashOverlay = findViewById<View>(R.id.splash_overlay)
        val splashRipple = findViewById<RippleView>(R.id.splash_ripple)
        val splashLogo = findViewById<ImageView>(R.id.splash_logo)

        splashLogo.animate()
            .alpha(1f)
            .scaleX(1f)
            .scaleY(1f)
            .setDuration(500)
            .start()

        val rippleRun = object : Runnable {
            override fun run() {
                if (splashRipple.width > 0) {
                    splashRipple.start(
                        splashRipple.width / 2f,
                        splashRipple.height / 2f
                    )
                }
                splashHandler.postDelayed(this, SPLASH_RIPPLE_INTERVAL_MS)
            }
        }
        splashRippleRun = rippleRun
        splashRipple.post { rippleRun.run() }

        splashHandler.postDelayed({
            splashRippleRun?.let { splashHandler.removeCallbacks(it) }
            splashOverlay.animate()
                .alpha(0f)
                .withLayer()
                .setDuration(SPLASH_FADE_MS)
                .withEndAction { splashOverlay.visibility = View.GONE }
                .start()
        }, SPLASH_DURATION_MS)
    }

    private fun wrapHeightForWidth(width: Int): Int {
        val cardH = (width * CARD_RATIO).toInt()

    @SuppressLint("ClickableViewAccessibility")
    private fun setupMiniPlayerGestures() {
        var startX = 0f
        var startY = 0f
        var startTouchX = 0f
        var startTouchY = 0f
        var isDragging = false
        var downTime = 0L

        var isPinching = false
        var pinchStartDist = 0f
        var pinchStartWidth = 0
        var pinchCenterX = 0f
        var pinchCenterY = 0f

        miniPlayerWrap.setOnTouchListener { v, event ->
            val parent = v.parent as? ViewGroup ?: return@setOnTouchListener false
            val parentW = parent.width.toFloat()
            val parentH = parent.height.toFloat()
            val marginF = marginPx.toFloat()
            val minWidthPx = (minWidthDp * density).toInt()
            val maxWidthPx = (maxWidthDp * density).toInt()

            when (event.actionMasked) {
                MotionEvent.ACTION_DOWN -> {
                    startX = v.x
                    startY = v.y
                    startTouchX = event.rawX
                    startTouchY = event.rawY
                    isDragging = false
                    downTime = System.currentTimeMillis()
                    true
                }
                MotionEvent.ACTION_POINTER_DOWN -> {
                    if (event.pointerCount >= 2) {
                        isPinching = true
                        isDragging = false
                        pinchStartDist = distanceBetween(event, 0, 1)
                        pinchStartWidth = v.width
                        pinchCenterX = v.x + v.width / 2f
                        pinchCenterY = v.y + v.height / 2f
                    }
                    true
                }
                MotionEvent.ACTION_MOVE -> {
                    if (isPinching && event.pointerCount >= 2) {
                        val curDist = distanceBetween(event, 0, 1)
                        if (pinchStartDist > 0f && pinchStartWidth > 0) {
                            val factor = curDist / pinchStartDist
                            var newW = (pinchStartWidth * factor).toInt()
                            newW = newW.coerceIn(minWidthPx, maxWidthPx)
                            val newH = wrapHeightForWidth(newW)
                            val maxX = (parentW - newW - marginPx).coerceAtLeast(marginF)
                            val maxY = (parentH - newH - marginPx).coerceAtLeast(marginF)
                            val lp = v.layoutParams
                            lp.width = newW
                            lp.height = newH
                            v.layoutParams = lp
                            v.x = (pinchCenterX - newW / 2f).coerceIn(marginF, maxX)
                            v.y = (pinchCenterY - newH / 2f).coerceIn(marginF, maxY)
                        }
                        return@setOnTouchListener true
                    }
                    val maxX = (parentW - v.width - marginPx).coerceAtLeast(marginF)
                    val maxY = (parentH - v.height - marginPx).coerceAtLeast(marginF)
                    val dx = event.rawX - startTouchX
                    val dy = event.rawY - startTouchY
                    if (abs(dx) > 10 || abs(dy) > 10) isDragging = true
                    v.x = (startX + dx).coerceIn(marginF, maxX)
                    v.y = (startY + dy).coerceIn(marginF, maxY)
                    true
                }
                MotionEvent.ACTION_POINTER_UP -> {
                    if (isPinching) {
                        val idx = event.actionIndex
                        val remainIdx = if (idx == 0) 1 else 0
                        if (remainIdx < event.pointerCount) {
                            startTouchX = event.getRawX(remainIdx)
                            startTouchY = event.getRawY(remainIdx)
                            startX = v.x
                            startY = v.y
                        }
                        isPinching = false
                        isDragging = false
                    }
                    true
                }
                MotionEvent.ACTION_UP -> {
                    if (!isDragging && System.currentTimeMillis() - downTime < 250) {
                        expandFromMini()
                    } else if (isDragging) {
                        val maxX = (parentW - v.width - marginPx).coerceAtLeast(marginF)
                        val maxY = (parentH - v.height - marginPx).coerceAtLeast(marginF)
                        val cx = v.x + v.width / 2f
                        val cy = v.y + v.height / 2f
                        val targetX = if (cx < parentW / 2f) marginF else maxX
                        val targetY = if (cy < parentH / 2f) marginF else maxY
                        v.animate().x(targetX).y(targetY).setDuration(200)
                            .withEndAction { refreshPipSourceRect() }.start()
                    }
                    isDragging = false
                    isPinching = false
                    refreshPipSourceRect()
                    true
                }
                MotionEvent.ACTION_CANCEL -> {
                    isDragging = false
                    isPinching = false
                    refreshPipSourceRect()
                    true
                }
                else -> false
            }
        }

        val blockDrag = View.OnTouchListener { btn, ev ->
            when (ev.actionMasked) {
                MotionEvent.ACTION_DOWN -> btn.parent?.requestDisallowInterceptTouchEvent(true)
                MotionEvent.ACTION_UP, MotionEvent.ACTION_CANCEL -> btn.parent?.requestDisallowInterceptTouchEvent(false)
            }
            false
        }
        miniBtnClose.setOnTouchListener(blockDrag)
        miniBtnPrev.setOnTouchListener(blockDrag)
        miniBtnPlayPause.setOnTouchListener(blockDrag)
        miniBtnNext.setOnTouchListener(blockDrag)
    }

    private fun distanceBetween(event: MotionEvent, i1: Int, i2: Int): Float {
        val dx = event.getX(i1) - event.getX(i2)
        val dy = event.getY(i1) - event.getY(i2)
        return hypot(dx, dy)
    }

    fun collapseToMini() {
        if (currentVideo == null || globalPlayer == null) return
        if (isAnimatingCollapse) return
        isAnimatingCollapse = true

        descriptionOverlayContainer.visibility = View.GONE
        commentsOverlayContainer.visibility = View.GONE

        val pf = getPlayerFragmentOrNull()
        val playerFrame = pf?.view?.findViewById<View>(R.id.player_frame)
        val playerRoot = pf?.view
        val contentView = getContentFragmentOrNull()?.view

        playerRoot?.animate()?.cancel()
        playerFrame?.animate()?.cancel()
        contentView?.animate()?.cancel()

        val targetW = (200 * density).toInt()
        val targetWrapH = wrapHeightForWidth(targetW)

        val overlayW = overlayContainer.width
        val overlayH = overlayContainer.height
        val targetX = (overlayW - targetW - marginPx).toFloat()
        val targetY = (overlayH - targetWrapH - marginPx).toFloat()

        val pfLoc = IntArray(2)
        playerFrame?.getLocationInWindow(pfLoc)
        val ovLoc = IntArray(2)
        overlayContainer.getLocationInWindow(ovLoc)
        val frameX = (pfLoc[0] - ovLoc[0]).toFloat()
        val frameY = (pfLoc[1] - ovLoc[1]).toFloat()
        val frameW = (playerFrame?.width ?: 1).toFloat().coerceAtLeast(1f)
        val frameH = (playerFrame?.height ?: 1).toFloat().coerceAtLeast(1f)

        val targetCardH = (targetW * CARD_RATIO).toInt()
        val scaleX = targetW / frameW
        val scaleY = targetCardH / frameH
        val dx = targetX - frameX - (frameW * (1 - scaleX) / 2f)
        val dy = targetY - frameY - (frameH * (1 - scaleY) / 2f)

        miniPlayerView.player = globalPlayer
        val lp = miniPlayerWrap.layoutParams
        lp.width = targetW
        lp.height = targetWrapH
        miniPlayerWrap.layoutParams = lp
        miniPlayerWrap.x = targetX
        miniPlayerWrap.y = targetY
        miniPlayerWrap.alpha = 0f
        miniPlayerWrap.visibility = View.VISIBLE

        playerFrame?.animate()
            ?.scaleX(scaleX)
            ?.scaleY(scaleY)
            ?.translationX(dx)
            ?.translationY(dy)
            ?.setDuration(300)
            ?.withEndAction {
                miniPlayerWrap.alpha = 1f
                playerRoot?.visibility = View.GONE
                playerFrame?.animate()?.cancel()
                playerFrame?.scaleX = 1f
                playerFrame?.scaleY = 1f
                playerFrame?.translationX = 0f
                playerFrame?.translationY = 0f
                playerRoot?.translationY = 0f

                removePlayerFragments()
                supportFragmentManager.findFragmentById(R.id.description_overlay_container)?.let {
                    supportFragmentManager.beginTransaction().remove(it).commit()
                }
                supportFragmentManager.findFragmentById(R.id.comments_overlay_container)?.let {
                    supportFragmentManager.beginTransaction().remove(it).commit()
                }

                playerContainer.visibility = View.GONE
                contentContainer.visibility = View.GONE
                overlayContainer.visibility = View.VISIBLE
                overlayContainer.translationY = 0f

    private fun loadPaletteFromThumbnail() {

        val url = video.thumbnail

        viewLifecycleOwner.lifecycleScope.launch {
            val bitmap: Bitmap? = withContext(Dispatchers.IO) {
                try {
                    val request = ImageRequest.Builder(requireContext())
                        .data(url)
                        .size(80, 80)
                        .allowHardware(false)
                        .build()
                    val result = requireContext().imageLoader.execute(request)
                    (result.drawable as? BitmapDrawable)?.bitmap
                } catch (_: Exception) {
                    null
                }
            }
            if (bitmap == null) return@launch

            val palette = withContext(Dispatchers.Default) {
                Palette.from(bitmap).generate()
            }

            val vibrant = palette.getVibrantColor(
                palette.getDominantColor(Color.BLACK)
            )
            val muted = palette.getMutedColor(
                palette.getDarkVibrantColor(Color.BLACK)
            )

            val darkVibrant = blendWithBlack(vibrant, PALETTE_BRIGHTNESS)
            val darkMuted = blendWithBlack(muted, PALETTE_BRIGHTNESS * 0.6f)

            // Красим playerFrame
            val gradient = android.graphics.drawable.GradientDrawable(
                android.graphics.drawable.GradientDrawable.Orientation.TOP_BOTTOM,

            )

            val current = (binding.playerFrame.background as? android.graphics.drawable.ColorDrawable)?.color
                ?: Color.BLACK
            val anim = ValueAnimator.ofObject(ArgbEvaluator(), current, darkVibrant).apply {
                duration = 400L
                addUpdateListener { a ->

                    binding.playerFrame.setBackgroundColor(c)
                }
                addListener(object : android.animation.AnimatorListenerAdapter() {
                    override fun onAnimationEnd(animation: android.animation.Animator) {
                        binding.playerFrame.background = gradient
                    }
                })
            }

    @SuppressLint("ClickableViewAccessibility")
    private fun setupTapAndSwipe() {
        var touchStartY = 0f
        var touchStartX = 0f
        var movedEnoughForSwipe = false
        var movedUpForFullscreen = false
        var movedBackSwipe = false
        var downInsidePlayer = false

        val backSwipeThresholdPx = (80 * resources.displayMetrics.density)

        binding.playerRoot.setOnTouchListener { _, event ->
            gestureDetector?.onTouchEvent(event)

            when (event.action) {
                MotionEvent.ACTION_DOWN -> {
                    touchStartY = event.rawY
                    touchStartX = event.rawX
                    movedEnoughForSwipe = false
                    movedUpForFullscreen = false
                    movedBackSwipe = false
                    downInsidePlayer = isPointInsidePlayerView(event)

                    if (downInsidePlayer) startLongPress()

                    false
                }
                MotionEvent.ACTION_MOVE -> {
                    val dy = event.rawY - touchStartY
                    val dx = event.rawX - touchStartX

                    if (abs(dy) > 20 || abs(dx) > 20) {
                        stopLongPress()
                    }

                    if (isFullscreen && dx > backSwipeThresholdPx && dx > abs(dy) * 1.5f) {
                        movedBackSwipe = true
                        return@setOnTouchListener true
                    }

                    if (dy < -50 && abs(dy) > abs(dx) * 1.5f && !isFullscreen) {
                        movedUpForFullscreen = true
                    }

                    if (dy > 30 && dy > abs(dx) * 1.5f) {
                        movedEnoughForSwipe = true
                        (requireActivity() as MainActivity).onPlayerSwipeDown(dy)
                    }

                    movedEnoughForSwipe || movedUpForFullscreen || movedBackSwipe
                }
                MotionEvent.ACTION_UP -> {
                    stopLongPress()

                    if (movedBackSwipe) {
                        handleBack()
                        true
                    } else if (movedUpForFullscreen && !isFullscreen) {
                        enterFullscreen()
                        true
                    } else if (movedEnoughForSwipe) {
                        val deltaY = event.rawY - touchStartY
                        if (deltaY > 150) {
                            val activity = requireActivity() as MainActivity
                            if (isFullscreen) {
                                exitFullscreen()
                                binding.playerView.postDelayed({
                                    activity.collapseToMini()
                                }, 250)
                            } else {
                                activity.collapseToMini()
                            }
                        } else {
                            (requireActivity() as MainActivity).onPlayerSwipeDown(0f)
                        }
                        true
                    } else {
                        false
                    }
                }
                MotionEvent.ACTION_CANCEL -> {
                    stopLongPress()
                    false
                }
                else -> false
            }
        }
    }

            exitFullscreen()
            return true

    }

                )

            override fun run() {
                if (_binding != null) {
                    updateTimes()
                    handler.postDelayed(this, 500)
                }
            }
        }
        handler.post(runnable)

        binding.btnPlayPause.setImageResource(
            if (player.isPlaying) R.drawable.ic_player_pause else R.drawable.ic_player_play

        if (_binding == null) return
        if (userSeeking) return
        val player = MainActivity.globalPlayer ?: return

        val duration = player.duration.coerceAtLeast(0L)
        val position = player.currentPosition.coerceAtLeast(0L)

        binding.currentTime.text = formatTime(position)
        binding.totalTime.text = formatTime(duration)

        if (duration > 0) {
            binding.progressBar.setProgress(position.toFloat() / duration.toFloat())
            binding.progressBar.setBuffered(player.bufferedPosition.toFloat() / duration.toFloat())
        } else {
            binding.progressBar.setProgress(0f)
            binding.progressBar.setBuffered(0f)
        }
    }

    private fun formatTime(ms: Long): String {
        if (ms <= 0) return "00:00"
        val totalSec = ms / 1000
        val h = totalSec / 3600
        val m = (totalSec % 3600) / 60
        val s = totalSec % 60
        return if (h > 0) {
            String.format("%d:%02d:%02d", h, m, s)
        } else {
            String.format("%02d:%02d", m, s)
        }
    }

    fun restoreAfterPip() { /* no-op */ }

package ru.asmr1.app.player

import android.content.Context
import android.graphics.Canvas
import android.graphics.Color
import android.graphics.Paint
import android.graphics.RectF
import android.util.AttributeSet
import android.view.MotionEvent
import android.view.View

/**
 * Только полоса прогресса. Thumb рисуется отдельным View поверх (thumb_view).
 * Сообщает координаты thumb через [onThumbPosition].
 */
class VideoProgressBar @JvmOverloads constructor(
    context: Context,
    attrs: AttributeSet? = null,
    defStyleAttr: Int = 0
) : View(context, attrs, defStyleAttr) {

    private var progress = 0f
    private var buffered = 0f
    private var expanded = false
    private var cornersEnabled = false

    private val thinHeightPx = dpToPx(2f)
    private val thickHeightPx = dpToPx(3.5f)
    private val cornerPx = dpToPx(2f)

    // Высота зоны кликов
    private val touchAreaPx = dpToPx(24f)

    private val bgColor = Color.parseColor("#66000000")
    private val progressColor = Color.WHITE

    private val bgPaint = Paint(Paint.ANTI_ALIAS_FLAG).apply {
        color = bgColor
        style = Paint.Style.FILL
    }
    private val progressPaint = Paint(Paint.ANTI_ALIAS_FLAG).apply {
        color = progressColor
        style = Paint.Style.FILL
    }

    private val tmpRect = RectF()

    var onSeek: ((Float) -> Unit)? = null
    var onSeekStart: (() -> Unit)? = null
    var onSeekEnd: (() -> Unit)? = null

    /** Callback для перемещения внешнего thumb: (x, y) в координатах этого View. */
    var onThumbPosition: ((Float, Float) -> Unit)? = null

    private var isDragging = false

    fun setProgress(p: Float) {
        progress = p.coerceIn(0f, 1f)
        invalidate()
        notifyThumbPosition()
    }

    fun setBuffered(b: Float) {
        buffered = b.coerceIn(0f, 1f)
        invalidate()
    }

    fun setExpanded(expand: Boolean) {
        expanded = expand
        invalidate()
    }

    fun setCornersEnabled(enabled: Boolean) {
        if (cornersEnabled == enabled) return
        cornersEnabled = enabled
        invalidate()
    }

    override fun onMeasure(widthMeasureSpec: Int, heightMeasureSpec: Int) {
        val desiredHeight = touchAreaPx.toInt()
        val h = resolveSize(desiredHeight, heightMeasureSpec)
        val w = resolveSize(suggestedMinimumWidth, widthMeasureSpec)
        setMeasuredDimension(w, h)
    }

    override fun onDraw(canvas: Canvas) {
        super.onDraw(canvas)

        val w = width.toFloat()
        val h = height.toFloat()

        val barH = if (expanded) thickHeightPx else thinHeightPx

        // Полоса в самом низу View
        val barBottom = h
        val barTop = h - barH

        val r = if (cornersEnabled) cornerPx else 0f

        // Фон полосы
        tmpRect.set(0f, barTop, w, barBottom)
        canvas.drawRoundRect(tmpRect, r, r, bgPaint)

        // Прогресс
        val progressW = w * progress
        if (progressW > 0f) {
            tmpRect.set(0f, barTop, progressW, barBottom)
            canvas.drawRoundRect(tmpRect, r, r, progressPaint)
        }

        // Thumb рисуется ВНЕШНИМ View, но координаты сообщаем отсюда
        notifyThumbPosition()
    }

    private fun notifyThumbPosition() {
        if (width == 0 || height == 0) return
        val barH = if (expanded) thickHeightPx else thinHeightPx
        val barBottom = height.toFloat()
        val barTop = barBottom - barH
        val cx = (width * progress).coerceIn(0f, width.toFloat())
        // Центр thumb на полосе
        val cy = barTop + barH / 2f
        onThumbPosition?.invoke(cx, cy)
    }

    override fun onTouchEvent(event: MotionEvent): Boolean {
        when (event.actionMasked) {
            MotionEvent.ACTION_DOWN -> {
                isDragging = true
                parent?.requestDisallowInterceptTouchEvent(true)
                onSeekStart?.invoke()
                handleTouch(event.x)
                return true
            }
            MotionEvent.ACTION_MOVE -> {
                if (isDragging) {
                    handleTouch(event.x)
                }
                return true
            }
            MotionEvent.ACTION_UP, MotionEvent.ACTION_CANCEL -> {
                if (isDragging) {
                    isDragging = false
                    parent?.requestDisallowInterceptTouchEvent(false)
                    onSeekEnd?.invoke()
                }
                return true
            }
        }
        return super.onTouchEvent(event)
    }

    private fun handleTouch(x: Float) {
        val p = (x / width).coerceIn(0f, 1f)
        progress = p
        invalidate()
        notifyThumbPosition()
        onSeek?.invoke(p)
    }

    fun setDragging(dragging: Boolean) {
        isDragging = dragging
    }

    private fun dpToPx(dp: Float): Float =
        dp * resources.displayMetrics.density
}

/**
 * Расширяющийся круг из точки нажатия, затухающий со временем.
 */
class RippleView @JvmOverloads constructor(
    context: Context,
    attrs: AttributeSet? = null,
    defStyleAttr: Int = 0
) : View(context, attrs, defStyleAttr) {

    private var centerX = 0f
    private var centerY = 0f
    private var radius = 0f
    private var maxRadius = 0f
    private var alpha = 0f

    private val paint = Paint(Paint.ANTI_ALIAS_FLAG).apply {
        style = Paint.Style.FILL
        color = Color.WHITE
    }

    private var animator: ValueAnimator? = null

    fun start(touchX: Float, touchY: Float) {
        centerX = touchX
        centerY = touchY
        maxRadius = Math.min(width.toFloat(), height.toFloat()) * 0.6f
        radius = 0f
        alpha = 0.35f

        animator?.cancel()
        animator = ValueAnimator.ofFloat(0f, 1f).apply {
            duration = 800
            interpolator = AccelerateDecelerateInterpolator()
            addUpdateListener { anim ->
                val v = anim.animatedValue as Float
                radius = maxRadius * v
                alpha = 0.35f * (1f - v)
                invalidate()
            }
            start()
        }
    }

    override fun onDraw(canvas: Canvas) {
        super.onDraw(canvas)
        if (radius <= 0f || alpha <= 0f) return
        paint.alpha = (alpha * 255).toInt().coerceIn(0, 255)
        canvas.drawCircle(centerX, centerY, radius, paint)
    }

    override fun onDetachedFromWindow() {
        super.onDetachedFromWindow()
        animator?.cancel()
        animator = null
    }
}

package ru.asmr1.app.player

import android.media.audiofx.LoudnessEnhancer
import android.os.Handler
import android.os.Looper

/**
 * Обёртка над LoudnessEnhancer.
 * - Усиливает звук до [targetGainMb] миллибел.
 * - Плавный fade-in с 0 до целевого усиления за [FADE_IN_MS] мс.
 */
class AudioBoost {

    companion object {
        // ×3 по амплитуде ≈ +9.5 дБ ≈ 950 mB
        private const val TARGET_GAIN_MB = 950
        private const val FADE_IN_MS = 1500L
        private const val FADE_STEP_MS = 30L
    }

    private var enhancer: LoudnessEnhancer? = null
    private var currentGainMb = 0
    private var fadeRunnable: Runnable? = null
    private val handler = Handler(Looper.getMainLooper())

    private var targetAudioSessionId: Int = 0

    fun attach(audioSessionId: Int) {
        if (audioSessionId == 0) return
        if (enhancer != null && targetAudioSessionId == audioSessionId) return

        release()
        targetAudioSessionId = audioSessionId

        try {
            enhancer = LoudnessEnhancer(audioSessionId).apply {
                setTargetGain(0)
                enabled = true
            }
        } catch (_: Exception) {
            enhancer = null
        }
    }

    /** Плавно увеличить громкость с 0 до TARGET_GAIN_MB за FADE_IN_MS. */
    fun fadeIn() {
        cancelFade()
        val e = enhancer ?: return

        currentGainMb = 0
        try {
            e.setTargetGain(0)
        } catch (_: Exception) {
        }

        val steps = (FADE_IN_MS / FADE_STEP_MS).toInt().coerceAtLeast(1)
        val stepGain = TARGET_GAIN_MB / steps
        var step = 0

        val r = object : Runnable {
            override fun run() {
                val e2 = enhancer ?: return
                step++
                currentGainMb = (currentGainMb + stepGain).coerceAtMost(TARGET_GAIN_MB)
                try {
                    e2.setTargetGain(currentGainMb)
                } catch (_: Exception) {
                }
                if (step < steps) {
                    handler.postDelayed(this, FADE_STEP_MS)
                }
            }
        }
        fadeRunnable = r
        handler.postDelayed(r, FADE_STEP_MS)
    }

    fun setGain(gainMb: Int) {
        cancelFade()
        val e = enhancer ?: return
        currentGainMb = gainMb.coerceIn(0, TARGET_GAIN_MB)
        try {
            e.setTargetGain(currentGainMb)
        } catch (_: Exception) {
        }
    }

    private fun cancelFade() {
        fadeRunnable?.let { handler.removeCallbacks(it) }
        fadeRunnable = null
    }

    fun release() {
        cancelFade()
        try {

package ru.asmr1.app.ui

import android.view.LayoutInflater
import android.view.View
import android.view.ViewGroup
import androidx.recyclerview.widget.DiffUtil
import androidx.recyclerview.widget.RecyclerView
import coil.imageLoader
import coil.load
import coil.request.CachePolicy
import coil.request.ImageRequest
import ru.asmr1.app.R
import ru.asmr1.app.data.Video
import ru.asmr1.app.databinding.ItemVideoBinding
import ru.asmr1.app.databinding.ItemVideoSkeletonBinding
import ru.asmr1.app.utils.formatDuration
import java.text.SimpleDateFormat
import java.util.Locale
import java.util.TimeZone

class VideoAdapter(
    private var videos: List<Video> = emptyList(),
    private val onClick: (Video) -> Unit = {},
    private val onMoreClick: (Video, View) -> Unit = { _, _ -> }
) : RecyclerView.Adapter<RecyclerView.ViewHolder>() {

    companion object {
        private const val TYPE_VIDEO = 0
        private const val TYPE_SKELETON = 1
        private const val SKELETON_COUNT = 5
        private const val PREFETCH_AHEAD = 2

        private val dateFmt = SimpleDateFormat("yyyy-MM-dd HH:mm:ss", Locale.US).apply {
            timeZone = TimeZone.getTimeZone("UTC")
        }
    }

    private var showSkeleton = false

    fun setVideos(newVideos: List<Video>) {
        val wasSkeleton = showSkeleton
        showSkeleton = false

        if (wasSkeleton) {
            videos = newVideos
            notifyDataSetChanged()
            return
        }

        val old = videos
        val diff = DiffUtil.calculateDiff(object : DiffUtil.Callback() {
            override fun getOldListSize() = old.size
            override fun getNewListSize() = newVideos.size
            override fun areItemsTheSame(o: Int, n: Int) = old[o].id == newVideos[n].id
            override fun areContentsTheSame(o: Int, n: Int) = old[o] == newVideos[n]
        })
        videos = newVideos
        diff.dispatchUpdatesTo(this)
    }

    fun appendVideos(newVideos: List<Video>) {
        if (newVideos.isEmpty()) return
        val startPos = videos.size
        videos = videos + newVideos
        notifyItemRangeInserted(startPos, newVideos.size)
    }

    fun showSkeleton() {
        showSkeleton = true
        videos = emptyList()
        notifyDataSetChanged()
    }

    override fun getItemViewType(position: Int): Int =
        if (showSkeleton) TYPE_SKELETON else TYPE_VIDEO

    override fun getItemCount(): Int =
        if (showSkeleton) SKELETON_COUNT else videos.size

    override fun onCreateViewHolder(parent: ViewGroup, viewType: Int): RecyclerView.ViewHolder {
        val inflater = LayoutInflater.from(parent.context)
        return when (viewType) {
            TYPE_SKELETON -> SkeletonViewHolder(
                ItemVideoSkeletonBinding.inflate(inflater, parent, false)
            )
            else -> VideoViewHolder(
                ItemVideoBinding.inflate(inflater, parent, false)
            )
        }
    }

    override fun onBindViewHolder(holder: RecyclerView.ViewHolder, position: Int) {
        when (holder) {
            is VideoViewHolder -> holder.bind(videos[position])
            is SkeletonViewHolder -> holder.bind()
        }
    }

    fun prefetchAround(context: android.content.Context, anchorPosition: Int) {
        if (showSkeleton) return
        val loader = context.imageLoader
        for (i in 1..PREFETCH_AHEAD) {
            val idx = anchorPosition + i
            if (idx >= videos.size) break
            val v = videos[idx]
            if (v.thumbnail.isNotEmpty()) {
                loader.enqueue(
                    ImageRequest.Builder(context)
                        .data(v.thumbnail)
                        .memoryCachePolicy(CachePolicy.ENABLED)
                        .diskCachePolicy(CachePolicy.ENABLED)
                        .build()
                )
            }
        }
    }

    inner class VideoViewHolder(private val binding: ItemVideoBinding) :
        RecyclerView.ViewHolder(binding.root) {

        fun bind(video: Video) {
            binding.title.text = video.title
            binding.meta.text = video.author.name

            binding.viewsOverlay.text = formatViewsShort(video.views)
            binding.dateOverlay.text = timeAgoShort(video.published)

            binding.thumbnail.transitionName = "video_thumb_" + video.id

            if (video.thumbnail.isNotEmpty()) {
                binding.thumbnail.load(video.thumbnail) {
                    crossfade(false)
                    placeholder(R.color.card_bg)
                    error(R.color.card_bg)
                    memoryCachePolicy(CachePolicy.ENABLED)
                    diskCachePolicy(CachePolicy.ENABLED)
                    allowHardware(true)
                }
            }

            if (video.author.avatar.isNotEmpty()) {
                binding.authorAvatar.load(video.author.avatar) {
                    crossfade(false)
                    placeholder(R.color.card_bg)
                    error(R.color.card_bg)
                    memoryCachePolicy(CachePolicy.ENABLED)
                    diskCachePolicy(CachePolicy.ENABLED)
                    allowHardware(true)
                }
            }

            if (video.duration > 0) {
                binding.duration.visibility = View.VISIBLE
                binding.duration.text = formatDuration(video.duration)
            } else {
                binding.duration.visibility = View.GONE
            }

            binding.root.setOnClickListener {
                val rect = android.graphics.Rect()
                binding.thumbnail.getGlobalVisibleRect(rect)
                onClick(video)
            }
            binding.btnMore.setOnClickListener { v -> onMoreClick(video, v) }
        }
    }

    class SkeletonViewHolder(private val binding: ItemVideoSkeletonBinding) :
        RecyclerView.ViewHolder(binding.root) {

        fun bind() {
            binding.veilLayout.veil()
        }
    }

    private fun formatViewsShort(n: Int): String = when {
        n >= 1_000_000 -> String.format(Locale.US, "%.1fM", n / 1_000_000.0)
        n >= 1_000 -> String.format(Locale.US, "%.1fK", n / 1_000.0)
        else -> n.toString()
    }

    private fun timeAgoShort(dateStr: String): String {
        if (dateStr.isEmpty()) return ""
        return try {
            val date = synchronized(dateFmt) { dateFmt.parse(dateStr) } ?: return ""
            val diff = (System.currentTimeMillis() - date.time) / 1000
            when {
                diff < 60 -> "сейчас"
                diff < 3600 -> "${diff / 60}м"
                diff < 86400 -> "${diff / 3600}ч"
                diff < 2_592_000 -> "${diff / 86400}д"
                diff < 31_536_000 -> "${diff / 2_592_000}мес"
                else -> "${diff / 31_536_000}г"
            }
        } catch (_: Exception) {
            ""
        }
    }
}

                isLoading = false
                _binding?.swipeRefresh?.isRefreshing = false
            }
        }

import android.os.Bundle

class FeedFragment : Fragment() {

    private var _binding: FragmentFeedBinding? = null
    private val binding get() = _binding!!

    private val tabs = listOf("recommended", "latest")

    override fun onCreateView(
        inflater: LayoutInflater,
        container: ViewGroup?,
        savedInstanceState: Bundle?
    ): View {
        _binding = FragmentFeedBinding.inflate(inflater, container, false)
        return binding.root
    }

    override fun onViewCreated(view: View, savedInstanceState: Bundle?) {
        super.onViewCreated(view, savedInstanceState)

        binding.viewPager.offscreenPageLimit = 1
        binding.viewPager.adapter = object : FragmentStateAdapter(this) {
            override fun getItemCount(): Int = tabs.size
            override fun createFragment(position: Int): Fragment =
                VideoListFragment.newInstance(tabs[position])
        }

        binding.viewPager.registerOnPageChangeCallback(object : ViewPager2.OnPageChangeCallback() {
            override fun onPageScrolled(position: Int, positionOffset: Float, positionOffsetPixels: Int) {
                val progress = position + positionOffset
                applyTabState(0, (1f - progress).coerceIn(0f, 1f))
                applyTabState(1, progress.coerceIn(0f, 1f))
            }

            override fun onPageSelected(position: Int) {
                // Сбрасываем скролл у СТАРОГО таба — того, с которого ушли
                val oldIndex = if (position == 0) 1 else 0
                getFragmentAt(oldIndex)?.scrollToTop()
            }
        })

        binding.tabRecommended.setOnClickListener {
            if (binding.viewPager.currentItem != 0) {
                binding.viewPager.setCurrentItem(0, true)
            }
        }
        binding.tabLatest.setOnClickListener {
            if (binding.viewPager.currentItem != 1) {
                binding.viewPager.setCurrentItem(1, true)
            }
        }

        applyTabState(0, 1f)
        applyTabState(1, 0f)
    }

    /**
     * ViewPager2 (FragmentStateAdapter) хранит фрагменты под тегами "f0", "f1", ...
     * Это единственный надёжный способ достать фрагмент по позиции.
     */
    private fun getFragmentAt(index: Int): VideoListFragment? {
        return childFragmentManager.findFragmentByTag("f$index") as? VideoListFragment
    }

    private fun applyTabState(index: Int, weight: Float) {
        val tab = if (index == 0) binding.tabRecommended else binding.tabLatest
        val scale = 0.94f + 0.06f * weight
        tab.scaleX = scale
        tab.scaleY = scale
        tab.alpha = 0.6f + 0.4f * weight
        tab.background?.mutate()?.alpha = (weight * 255).toInt().coerceIn(0, 255)
    }

    override fun onDestroyView() {
        super.onDestroyView()
        _binding = null
    }
}

    /** Если есть активный запрос — перезапускаем поиск с новой сортировкой */
    private fun applySortAndSearch() {
        val query = binding.etSearch.text?.toString()?.trim().orEmpty()
        if (query.isNotEmpty()) {
            currentQuery = query
            currentPage = 1
            hasMore = true
            performSearch(query, 1, append = false)
        }
    }

    private fun performSearch(query: String, page: Int, append: Boolean) {
        if (isLoading) return
        if (query.isEmpty()) return

        isLoading = true

        if (!append) {
            showSkeletons()
            _binding?.apply {
                llHistory.visibility = View.GONE
                llEmpty.visibility = View.GONE
                llNoResults.visibility = View.GONE
                swipeRefresh.visibility = View.VISIBLE
                llSortBar.visibility = View.VISIBLE
            }
        }

        searchJob?.cancel()
        searchJob = viewLifecycleOwner.lifecycleScope.launch {
            try {
                val response = ApiClient.service.search(
                    query = query,
                    page = page,
                    perPage = perPage,
                    orderBy = orderBy,
                    order = order
                )

                if (_binding == null) return@launch

                if (response.success) {
                    if (append) {
                        adapter.appendVideos(response.videos)
                    } else {
                        adapter.setVideos(response.videos)
                    }
                    hasMore = response.hasMore

                    if (response.videos.isEmpty()) {
                        showNoResults(query)
                    } else {
                        showResults()
                    }
                } else {
                    if (!append) showNoResults(query)
                }
            } catch (e: Exception) {
                if (_binding == null) return@launch
                Toast.makeText(context, "Ошибка поиска: ${e.message}", Toast.LENGTH_SHORT).show()
                if (!append) showNoResults(query)
            } finally {
                isLoading = false
                _binding?.swipeRefresh?.isRefreshing = false
            }
        }
    }

    private fun showInitialState() {

            swipeRefresh.visibility = View.GONE
            llNoResults.visibility = View.GONE

        private const val TYPE_VIDEO = 0
        private const val TYPE_SKELETON = 1
        private const val SKELETON_COUNT = 6

        // Общий SimpleDateFormat (с UTC)
        private val dateFmt = SimpleDateFormat("yyyy-MM-dd HH:mm:ss", Locale.US).apply {
            timeZone = TimeZone.getTimeZone("UTC")
        }
    }

    private var showSkeleton = false

    fun setVideos(newVideos: List<Video>) {
        val wasSkeleton = showSkeleton
        showSkeleton = false

        if (wasSkeleton) {
            videos = newVideos
            notifyDataSetChanged()
            return
        }

        val old = videos
        val diff = DiffUtil.calculateDiff(object : DiffUtil.Callback() {
            override fun getOldListSize() = old.size
            override fun getNewListSize() = newVideos.size
            override fun areItemsTheSame(o: Int, n: Int) = old[o].id == newVideos[n].id
            override fun areContentsTheSame(o: Int, n: Int) = old[o] == newVideos[n]
        })
        videos = newVideos
        diff.dispatchUpdatesTo(this)
    }

    fun appendVideos(newVideos: List<Video>) {
        if (newVideos.isEmpty()) return
        val startPos = videos.size
        videos = videos + newVideos
        notifyItemRangeInserted(startPos, newVideos.size)
    }

    fun showSkeletons() {
        showSkeleton = true
        videos = emptyList()
        notifyDataSetChanged()
    }

    fun clear() {
        showSkeleton = false
        videos = emptyList()
        notifyDataSetChanged()
    }

    override fun getItemViewType(position: Int): Int =
        if (showSkeleton) TYPE_SKELETON else TYPE_VIDEO

    override fun getItemCount(): Int =
        if (showSkeleton) SKELETON_COUNT else videos.size

    override fun onCreateViewHolder(parent: ViewGroup, viewType: Int): RecyclerView.ViewHolder {
        val inflater = LayoutInflater.from(parent.context)
        return when (viewType) {
            TYPE_SKELETON -> SkeletonViewHolder(

            )
            else -> VideoViewHolder(
                ItemVideoSearchBinding.inflate(inflater, parent, false)

            is VideoViewHolder -> holder.bind(videos[position])
            is SkeletonViewHolder -> holder.bind()
        }
    }

    private fun formatViewsShort(n: Int): String = when {
        n >= 1_000_000 -> String.format(Locale.US, "%.1fM", n / 1_000_000.0)
        n >= 1_000 -> String.format(Locale.US, "%.1fK", n / 1_000.0)
        else -> n.toString()
    }

    private fun timeAgoShort(dateStr: String): String {
        if (dateStr.isEmpty()) return ""
        return try {
            val date = synchronized(dateFmt) { dateFmt.parse(dateStr) } ?: return ""
            val diff = (System.currentTimeMillis() - date.time) / 1000
            when {
                diff < 60 -> "сейчас"
                diff < 3600 -> "${diff / 60} мин"
                diff < 86400 -> "${diff / 3600} ч"
                diff < 2_592_000 -> "${diff / 86400} дн"
                diff < 31_536_000 -> "${diff / 2_592_000} мес"
                else -> "${diff / 31_536_000} г"
            }
        } catch (_: Exception) {
            ""
        }
    }

                    error(R.color.card_bg)
                    memoryCachePolicy(CachePolicy.ENABLED)
                    diskCachePolicy(CachePolicy.ENABLED)
                    allowHardware(true)
                }
            }

            if (video.duration > 0) {
                binding.duration.visibility = View.VISIBLE
                binding.duration.text = formatDuration(video.duration)
            } else {
                binding.duration.visibility = View.GONE
            }

            binding.root.setOnClickListener { onClick(video) }
        }
    }

package ru.asmr1.app.auth

import android.content.Context
import androidx.security.crypto.EncryptedSharedPreferences
import androidx.security.crypto.MasterKey

class AuthManager(context: Context) {

    private val masterKey = MasterKey.Builder(context)
        .setKeyScheme(MasterKey.KeyScheme.AES256_GCM)
        .build()

    private val prefs = EncryptedSharedPreferences.create(
        context,
        "asmr1_auth_prefs",
        masterKey,
        EncryptedSharedPreferences.PrefKeyEncryptionScheme.AES256_SIV,
        EncryptedSharedPreferences.PrefValueEncryptionScheme.AES256_GCM
    )

    fun saveAuth(
        token: String,
        username: String,
        displayName: String,
        email: String,
        userId: Int
    ) {
        prefs.edit()
            .putString(KEY_TOKEN, token)
            .putString(KEY_USERNAME, username)
            .putString(KEY_DISPLAY_NAME, displayName)
            .putString(KEY_EMAIL, email)
            .putInt(KEY_USER_ID, userId)
            .putLong(KEY_LOGIN_TIME, System.currentTimeMillis())
            .apply()
    }

    fun getToken(): String? = prefs.getString(KEY_TOKEN, null)
    fun getUsername(): String? = prefs.getString(KEY_USERNAME, null)
    fun getDisplayName(): String? = prefs.getString(KEY_DISPLAY_NAME, null)
    fun getEmail(): String? = prefs.getString(KEY_EMAIL, null)
    fun getUserId(): Int = prefs.getInt(KEY_USER_ID, 0)
    fun getLoginTime(): Long = prefs.getLong(KEY_LOGIN_TIME, 0L)

    fun isLoggedIn(): Boolean = !getToken().isNullOrEmpty()

    fun clear() {
        prefs.edit().clear().apply()
    }

    companion object {
        private const val KEY_TOKEN = "jwt_token"
        private const val KEY_USERNAME = "jwt_username"
        private const val KEY_DISPLAY_NAME = "jwt_display_name"
        private const val KEY_EMAIL = "jwt_email"
        private const val KEY_USER_ID = "jwt_user_id"
        private const val KEY_LOGIN_TIME = "jwt_login_time"
    }
}

class LoginFragment : Fragment() {

    private var _binding: FragmentLoginBinding? = null
    private val binding get() = _binding!!

    private lateinit var authManager: AuthManager

    override fun onCreateView(
        inflater: LayoutInflater,
        container: ViewGroup?,
        savedInstanceState: Bundle?
    ): View {
        _binding = FragmentLoginBinding.inflate(inflater, container, false)
        return binding.root
    }

    override fun onViewCreated(view: View, savedInstanceState: Bundle?) {
        super.onViewCreated(view, savedInstanceState)
        authManager = AuthManager(requireContext())

        binding.btnLogin.setOnClickListener { performLogin() }
        binding.btnGuest.setOnClickListener { closeLogin() }
        binding.btnClose.setOnClickListener { closeLogin() }
        binding.btnRegister.setOnClickListener {
            (activity as? MainActivity)?.showRegisterScreen()
        }
    }

    private fun performLogin() {
        val username = binding.etUsername.text?.toString()?.trim().orEmpty()
        val password = binding.etPassword.text?.toString()?.trim().orEmpty()

        if (username.isEmpty() || password.isEmpty()) {
            toast("Заполните все поля")
            return
        }

        showLoading(true)

        viewLifecycleOwner.lifecycleScope.launch {
            try {
                val response = ApiClient.authApi.login(
                    LoginRequest(username = username, password = password)
                )
                val token = response.token

                authManager.saveAuth(
                    token = token,
                    username = response.userNicename,
                    displayName = response.userDisplayName,
                    email = response.userEmail,
                    userId = 0
                )

                try {
                    val userApi = ApiClient.createJwtAuthService(token)
                    val user = userApi.getCurrentUser()
                    authManager.saveAuth(
                        token = token,
                        username = user.slug ?: response.userNicename,
                        displayName = user.name,
                        email = response.userEmail,
                        userId = user.id
                    )
                } catch (_: Exception) { }

                toast("Добро пожаловать, ${response.userDisplayName}!")
                (activity as? MainActivity)?.onLoginSuccess()

            } catch (e: Exception) {
                showLoading(false)
                val msg = e.message ?: "Неизвестная ошибка"
                val cleanMsg = when {
                    msg.contains("401") || msg.contains("403") -> "Неверный логин или пароль"
                    msg.contains("timeout", true) -> "Сервер не отвечает"
                    else -> "Ошибка: $msg"
                }
                toast(cleanMsg)
            }
        }
    }

    private fun showLoading(show: Boolean) {
        _binding?.apply {
            progressBar.visibility = if (show) View.VISIBLE else View.GONE
            btnLogin.isEnabled = !show
            btnRegister.isEnabled = !show
            btnGuest.isEnabled = !show
            etUsername.isEnabled = !show
            etPassword.isEnabled = !show
        }
    }

    private fun closeLogin() {
        parentFragmentManager.popBackStack()
    }

    private fun toast(msg: String) {
        Toast.makeText(context, msg, Toast.LENGTH_SHORT).show()
    }

    override fun onDestroyView() {
        super.onDestroyView()
        _binding = null
    }
}

fun formatViews(n: Int): String = when {
    n >= 1_000_000 -> String.format("%.1fM просмотров", n / 1_000_000.0)
    n >= 1_000 -> String.format("%.1fK просмотров", n / 1_000.0)
    else -> "$n просмотров"
}

fun formatDuration(sec: Int): String {
    val m = sec / 60
    val s = sec % 60
    return "$m:${if (s < 10) "0$s" else s}"
}

fun timeAgo(dateStr: String): String {
    if (dateStr.isEmpty()) return ""
    return try {
        val fmt = SimpleDateFormat("yyyy-MM-dd HH:mm:ss", Locale.US).apply {
            timeZone = TimeZone.getTimeZone("UTC")
        }
        val date = fmt.parse(dateStr) ?: return ""
        val diff = (System.currentTimeMillis() - date.time) / 1000
        when {
            diff < 60 -> "только что"
            diff < 3600 -> "${diff / 60} мин назад"
            diff < 86400 -> "${diff / 3600} ч назад"
            diff < 2592000 -> "${diff / 86400} дн назад"
            diff < 31536000 -> "${diff / 2592000} мес назад"
            else -> "${diff / 31536000} г назад"
        }
    } catch (_: Exception) {
        ""
    }
}

package ru.asmr1.app.utils

import android.content.Context

/**
 * Локальная история поиска — сохраняется в SharedPreferences.
 * Максимум 20 запросов, новые сверху, без дубликатов.
 */
class SearchHistoryManager(context: Context) {

    private val prefs = context.getSharedPreferences("asmr1_search_history", Context.MODE_PRIVATE)

    companion object {
        private const val KEY_HISTORY = "history_items"
        private const val SEPARATOR = "|||"
        private const val MAX_ITEMS = 20
    }

    fun getHistory(): List<String> {
        val raw = prefs.getString(KEY_HISTORY, null) ?: return emptyList()
        if (raw.isEmpty()) return emptyList()
        return raw.split(SEPARATOR).filter { it.isNotBlank() }
    }

    fun addQuery(query: String) {
        val trimmed = query.trim()
        if (trimmed.isEmpty()) return

        val current = getHistory().toMutableList()
        current.removeAll { it.equals(trimmed, ignoreCase = true) }
        current.add(0, trimmed)
        val limited = current.take(MAX_ITEMS)
        prefs.edit().putString(KEY_HISTORY, limited.joinToString(SEPARATOR)).apply()
    }

    fun removeQuery(query: String) {
        val current = getHistory().toMutableList()
        current.removeAll { it == query }
        prefs.edit().putString(KEY_HISTORY, current.joinToString(SEPARATOR)).apply()
    }

    fun clearAll() {
        prefs.edit().remove(KEY_HISTORY).apply()
    }

    fun isEmpty(): Boolean = getHistory().isEmpty()
}
