    @SuppressLint("ClickableViewAccessibility")
    private fun setupMiniPlayerGestures() {
        var startX = 0f
        var startY = 0f
        var startTouchX = 0f
        var startTouchY = 0f
        var isDragging = false
        var downTime = 0L

        miniPlayerContainer.setOnTouchListener { v, event ->
            val parentW = (v.parent as ViewGroup).width.toFloat()
            val parentH = (v.parent as ViewGroup).height.toFloat()
            val marginF = marginPx.toFloat()
            val maxX = (parentW - v.width - marginPx).coerceAtLeast(marginF)
            val maxY = (parentH - v.height - marginPx).coerceAtLeast(marginF)

            when (event.actionMasked) {
                MotionEvent.ACTION_DOWN -> {
                    startX = v.x
                    startY = v.y
                    startTouchX = event.rawX
                    startTouchY = event.rawY
                    isDragging = false
                    downTime = System.currentTimeMillis()
                    true
                }
                MotionEvent.ACTION_MOVE -> {
                    val dx = event.rawX - startTouchX
                    val dy = event.rawY - startTouchY
                    if (abs(dx) > 10 || abs(dy) > 10) isDragging = true

                    v.x = (startX + dx).coerceIn(marginF, maxX)
                    v.y = (startY + dy).coerceIn(marginF, maxY)
                    true
                }
                MotionEvent.ACTION_UP -> {
                    val elapsed = System.currentTimeMillis() - downTime
                    if (!isDragging && elapsed < 250) {
                        expandFromMini()
                    } else if (isDragging) {
                        val cx = v.x + v.width / 2f
                        val cy = v.y + v.height / 2f
                        val targetX = if (cx < parentW / 2f) marginF else maxX
                        val targetY = if (cy < parentH / 2f) marginF else maxY

                        v.animate()
                            .x(targetX)
                            .y(targetY)
                            .setDuration(200)
                            .withEndAction { refreshPipSourceRect() }
                            .start()
                    }
                    isDragging = false
                    true
                }
                else -> false
            }
        }
    }

    private fun getPlayerFragmentOrNull(): PlayerFragment? {

    fun openPlayer(video: Video) {
        currentVideo = video

        globalPlayer?.apply {
            stop()
            clearMediaItems()
            setMediaItem(MediaItem.fromUri(video.videoUrl))
            prepare()
            playWhenReady = true
        }

        overlayContainer.visibility = View.VISIBLE
        overlayContainer.translationY = overlayContainer.height.toFloat()
        overlayContainer.animate()
            .translationY(0f)
            .setDuration(250)
            .start()

        playerContainer.visibility = View.VISIBLE
        contentContainer.visibility = View.VISIBLE
        miniPlayerContainer.visibility = View.GONE
        miniPlayerView.player = null

        supportFragmentManager.beginTransaction()
            .replace(R.id.player_container, PlayerFragment.newInstance(video))
            .commit()

        enablePip()
    }

    fun collapseToMini() {
        if (currentVideo == null || globalPlayer == null) return

        supportFragmentManager.findFragmentById(R.id.player_container)?.let { f ->
            supportFragmentManager.beginTransaction().remove(f).commit()
        }

        playerContainer.visibility = View.GONE
        contentContainer.visibility = View.GONE
        overlayContainer.visibility = View.VISIBLE
        overlayContainer.translationY = 0f

        miniPlayerView.player = globalPlayer
        miniPlayerContainer.visibility = View.VISIBLE

        miniPlayerContainer.post {
            val parentW = (miniPlayerContainer.parent as ViewGroup).width
            val parentH = (miniPlayerContainer.parent as ViewGroup).height
            miniPlayerContainer.x = (parentW - miniPlayerContainer.width - marginPx).toFloat()
            miniPlayerContainer.y = (parentH - miniPlayerContainer.height - marginPx).toFloat()
            refreshPipSourceRect()
        }

        miniPlayerContainer.alpha = 0f
        miniPlayerContainer.animate().alpha(1f).setDuration(200).start()

        feedContainer.visibility = View.VISIBLE
    }

    fun expandFromMini() {
        val video = currentVideo ?: return

        miniPlayerContainer.animate()
            .alpha(0f)
            .setDuration(150)
            .withEndAction {
                miniPlayerContainer.visibility = View.GONE
                miniPlayerView.player = null

                playerContainer.visibility = View.VISIBLE
                contentContainer.visibility = View.VISIBLE

                supportFragmentManager.beginTransaction()
                    .replace(R.id.player_container, PlayerFragment.newInstance(video))
                    .commit()

                playerContainer.postDelayed({
                    getPlayerFragmentOrNull()?.restoreAfterPip()
                }, 100)

                enablePip()
            }
            .start()
    }

    fun closePlayer() {
        globalPlayer?.stop()
        globalPlayer?.clearMediaItems()
        currentVideo = null

        miniPlayerContainer.visibility = View.GONE
        miniPlayerView.player = null
        overlayContainer.visibility = View.GONE
        feedContainer.visibility = View.VISIBLE

        disablePip()
    }

    fun setOverlayOffset(offset: Float) {
        overlayContainer.translationY = offset
    }

    private fun enablePip() {
        if (Build.VERSION.SDK_INT < Build.VERSION_CODES.S) return

        try {
            val builder = PictureInPictureParams.Builder()
                .setAspectRatio(Rational(16, 9))
                .setAutoEnterEnabled(true)
                .setSeamlessResizeEnabled(true)

            if (miniPlayerContainer.visibility == View.VISIBLE &&
                miniPlayerContainer.width > 0 &&
                miniPlayerContainer.height > 0
            ) {
                val rect = Rect()
                miniPlayerContainer.getGlobalVisibleRect(rect)
                if (!rect.isEmpty) {
                    builder.setSourceRectHint(rect)
                }
            }

            setPictureInPictureParams(builder.build())
        } catch (_: Exception) {
        }
    }

    private fun disablePip() {
        if (Build.VERSION.SDK_INT < Build.VERSION_CODES.S) return

        try {
            val params = PictureInPictureParams.Builder()
                .setAutoEnterEnabled(false)
                .build()
            setPictureInPictureParams(params)
        } catch (_: Exception) {
        }
    }

    private fun refreshPipSourceRect() {
        if (miniPlayerContainer.visibility != View.VISIBLE) return
        if (miniPlayerContainer.width == 0 || miniPlayerContainer.height == 0) return
        if (Build.VERSION.SDK_INT < Build.VERSION_CODES.S) return

        try {
            val rect = Rect()
            miniPlayerContainer.getGlobalVisibleRect(rect)
            if (rect.isEmpty) return

            val params = PictureInPictureParams.Builder()
                .setAspectRatio(Rational(16, 9))
                .setAutoEnterEnabled(true)
                .setSeamlessResizeEnabled(true)
                .setSourceRectHint(rect)
                .build()
            setPictureInPictureParams(params)
        } catch (_: Exception) {
        }
    }

    // ============================================================
    // МИНИ-ПЛЕЕР: ПРИНУДИТЕЛЬНЫЙ СБРОС
    // ============================================================
    private fun forceResetMiniPlayer() {

        if (miniPlayerRestoreW > 0) {
            val lp = miniPlayerContainer.layoutParams
            lp.width = miniPlayerRestoreW
            lp.height = miniPlayerRestoreH
            miniPlayerContainer.layoutParams = lp
            miniPlayerContainer.x = miniPlayerRestoreX
            miniPlayerContainer.y = miniPlayerRestoreY
        } else {
            val lp = miniPlayerContainer.layoutParams
            if (lp.width == ViewGroup.LayoutParams.MATCH_PARENT ||
                lp.height == ViewGroup.LayoutParams.MATCH_PARENT
            ) {
                lp.width = (200 * density).toInt()
                lp.height = (113 * density).toInt()
                miniPlayerContainer.layoutParams = lp
            }
        }
        miniPlayerRestoreW = 0
        miniPlayerRestoreH = 0

        miniPlayerContainer.animate().cancel()
        miniPlayerContainer.visibility = View.GONE
        miniPlayerContainer.alpha = 0f

        if (miniPlayerView.player == globalPlayer) {
            miniPlayerView.player = null
        }
    }

    private fun restoreBigPlayerPaused() {

        if (currentVideo == null) {
            return
        }

        forceResetMiniPlayer()
        getPlayerFragmentOrNull()?.restoreAfterPip()

        feedContainer.visibility = View.VISIBLE
        overlayContainer.visibility = View.VISIBLE
        overlayContainer.translationY = 0f
        playerContainer.visibility = View.VISIBLE
        contentContainer.visibility = View.VISIBLE

        val hasPlayerFragment = supportFragmentManager
            .findFragmentById(R.id.player_container) is PlayerFragment

        if (!hasPlayerFragment) {
            supportFragmentManager.beginTransaction()
                .replace(
                    R.id.player_container,
                    PlayerFragment.newInstance(currentVideo!!)
                )
                .commitAllowingStateLoss()
        }
    }

    override fun onUserLeaveHint() {
        super.onUserLeaveHint()
        getPlayerFragmentOrNull()?.expandForPip()
        refreshPipSourceRect()
    }

    override fun onPictureInPictureModeChanged(
        isInPictureInPictureMode: Boolean,
        newConfig: Configuration
    ) {
        super.onPictureInPictureModeChanged(isInPictureInPictureMode, newConfig)

        if (isInPictureInPictureMode) {
            wasInPip = true

            feedContainer.visibility = View.GONE
            contentContainer.visibility = View.GONE

            if (miniPlayerContainer.visibility == View.VISIBLE) {
                miniPlayerRestoreW = miniPlayerContainer.width
                miniPlayerRestoreH = miniPlayerContainer.height
                miniPlayerRestoreX = miniPlayerContainer.x
                miniPlayerRestoreY = miniPlayerContainer.y

                miniPlayerContainer.animate().cancel()
                miniPlayerContainer.alpha = 1f

                val lp = miniPlayerContainer.layoutParams
                lp.width = ViewGroup.LayoutParams.MATCH_PARENT
                lp.height = ViewGroup.LayoutParams.MATCH_PARENT
                miniPlayerContainer.layoutParams = lp
                miniPlayerContainer.x = 0f
                miniPlayerContainer.y = 0f
            }
        } else {
            if (!wasInPip) {
                return
            }
            wasInPip = false

            globalPlayer?.pause()
            forceResetMiniPlayer()
            pendingRestoreModal = true


            // Пытаемся понять — разворот это или закрытие:
            //  * Разворот тапом: onStop НЕ вызывается. Значит через 300мс
            //    pendingRestoreModal всё ещё true → восстанавливаем сами.
            //  * Закрытие крестиком: onStop вызовется и сам сбросит флаг через
            //    onStart, поэтому здесь ничего не делаем.
            android.os.Handler(android.os.Looper.getMainLooper()).postDelayed({
                if (pendingRestoreModal) {
                    pendingRestoreModal = false
                    restoreBigPlayerPaused()
                    enablePip()
                } else {
                }
            }, 300)
        }
    }

    override fun onStop() {
        super.onStop()

        if (wasInPip) {
            wasInPip = false
            globalPlayer?.pause()
            pendingRestoreModal = true
        }

        if (miniPlayerRestoreW > 0 && miniPlayerContainer.visibility == View.VISIBLE) {
            val lp = miniPlayerContainer.layoutParams
            lp.width = miniPlayerRestoreW
            lp.height = miniPlayerRestoreH
            miniPlayerContainer.layoutParams = lp
            miniPlayerContainer.x = miniPlayerRestoreX
            miniPlayerContainer.y = miniPlayerRestoreY
        }
        miniPlayerRestoreW = 0
        miniPlayerRestoreH = 0
    }

    override fun onStart() {
        super.onStart()

        if (pendingRestoreModal) {
            pendingRestoreModal = false

            if (currentVideo != null) {
                window.decorView.post {
                    restoreBigPlayerPaused()
                    enablePip()
                }
            }
        }
    }

    override fun onDestroy() {
        super.onDestroy()
        if (isFinishing) {
            globalPlayer?.release()
            globalPlayer = null
            currentVideo = null
        }
    }
}

    private var _binding: FragmentPlayerBinding? = null

    override fun onCreateView(
        inflater: LayoutInflater,
        container: ViewGroup?,
        savedInstanceState: Bundle?
    ): View {
        _binding = FragmentPlayerBinding.inflate(inflater, container, false)

    }

    @SuppressLint("ClickableViewAccessibility")
    override fun onViewCreated(view: View, savedInstanceState: Bundle?) {
        super.onViewCreated(view, savedInstanceState)

        val video = requireArguments().getParcelable<Video>(ARG_VIDEO) ?: return

        binding.title.text = video.title
        binding.authorName.text = video.author.name
        binding.meta.text = "${formatViews(video.views)} • ${timeAgo(video.published)}"
        binding.description.text = video.description

        if (video.author.avatar.isNotEmpty()) {
            binding.authorAvatar.load(video.author.avatar)
        }

        val activity = requireActivity() as MainActivity
        binding.playerView.player = MainActivity.globalPlayer

        binding.btnClose.setOnClickListener { activity.closePlayer() }

        // Свайп вниз ТОЛЬКО по верхней полоске над плеером (не по плееру)
        // В YouTube — свайп по мини "хендлу" сверху плеера.
        // Проще: вешаем на ImageView-poster + отдельная полоса над плеером.
        // Здесь вешаем на btn_close-зону и на верхние 40dp плеера.

        binding.playerView.setOnTouchListener { _, event ->
            when (event.action) {
                MotionEvent.ACTION_DOWN -> {
                    touchStartY = event.rawY
                    isDragging = false
                    // Не перехватываем клик — пусть контролы работают
                    false
                }
                MotionEvent.ACTION_MOVE -> {
                    val deltaY = event.rawY - touchStartY
                    if (deltaY > 60) {
                        isDragging = true
                        binding.root.translationY = deltaY
                        binding.root.alpha = 1f - (deltaY / 1000f).coerceIn(0f, 1f)
                    }
                    isDragging
                }
                MotionEvent.ACTION_UP -> {
                    if (isDragging) {
                        val deltaY = event.rawY - touchStartY
                        if (deltaY > 150) {
                            binding.root.animate()
                                .translationY(binding.root.height.toFloat())
                                .alpha(0f)
                                .setDuration(200)
                                .withEndAction { activity.collapseToMini() }
                                .start()
                        } else {
                            binding.root.animate()
                                .translationY(0f)
                                .alpha(1f)
                                .setDuration(200)
                                .start()
                        }
                        isDragging = false
                        true
                    } else {
                        false
                    }
                }
                else -> false
            }
        }
    }

    // ============================================================
    // PiP: раздуть плеер на весь экран и вернуть обратно
    // ============================================================


    // ============================================================
    // PiP: раздуть плеер на весь экран и вернуть обратно
    // ============================================================

    fun expandForPip() {
        if (isExpandedForPip) return
        val root = view ?: return
        if (!isAdded) return

        // Запоминаем исходные размеры
        originalPlayerHeight = binding.playerView.layoutParams.height

        val activity = requireActivity()
        val screenWidth = activity.resources.displayMetrics.widthPixels
        val targetHeight = (screenWidth * 9 / 16f).toInt()

        // 1. Корень фрагмента — на весь экран (чтобы не было чёрных полос)
        root.layoutParams = root.layoutParams.apply {
            height = ViewGroup.LayoutParams.MATCH_PARENT
        }

        // 2. Чёрный контейнер — ровно 16:9, прижат к верху
        binding.playerFrame.layoutParams = binding.playerFrame.layoutParams.apply {
            height = targetHeight
        }

        // 3. PlayerView — ровно 16:9
        binding.playerView.layoutParams = binding.playerView.layoutParams.apply {
            height = targetHeight
        }

        // 4. Скрыть всё лишнее
        binding.thumbnail.visibility = View.GONE
        binding.btnClose.visibility = View.GONE
        binding.playerScroll.visibility = View.GONE

        isExpandedForPip = true
    }

    fun restoreAfterPip() {
        if (!isExpandedForPip) return
        val root = view ?: return
        if (!isAdded) return

        // 1. Вернуть корень фрагмента
        root.layoutParams = root.layoutParams.apply {
            height = ViewGroup.LayoutParams.MATCH_PARENT
        }

        // 2. Вернуть чёрный контейнер (wrap_content)
        binding.playerFrame.layoutParams = binding.playerFrame.layoutParams.apply {
            height = ViewGroup.LayoutParams.WRAP_CONTENT
        }

        // 3. Вернуть высоту PlayerView
        binding.playerView.layoutParams = binding.playerView.layoutParams.apply {
            height = if (originalPlayerHeight > 0) originalPlayerHeight
                     else (220 * resources.displayMetrics.density).toInt()
        }

        // 4. Показать обратно
        binding.thumbnail.visibility = View.VISIBLE
        binding.btnClose.visibility = View.VISIBLE
        binding.playerScroll.visibility = View.VISIBLE

        isExpandedForPip = false
    }

    override fun onDestroyView() {
        super.onDestroyView()
        binding.playerView.player = null
        _binding = null
    }
}

    override fun onCreateView(
        inflater: LayoutInflater,
        container: ViewGroup?,
        savedInstanceState: Bundle?
    ): View {
        _binding = FragmentFeedBinding.inflate(inflater, container, false)
        return binding.root
    }

    override fun onViewCreated(view: View, savedInstanceState: Bundle?) {
        super.onViewCreated(view, savedInstanceState)

        setupRecyclerView()
        setupTabs()
        setupSwipeRefresh()
        setupInfiniteScroll()
        loadFeed(currentTab, 1, false)
    }

    private fun setupRecyclerView() {
        adapter = VideoAdapter { video ->
            openVideo(video)
        }
        layoutManager = LinearLayoutManager(requireContext())
        binding.recyclerView.layoutManager = layoutManager
        binding.recyclerView.adapter = adapter
    }

    private fun setupTabs() {
        binding.tabRecommended.setOnClickListener {
            if (currentTab == "recommended") return@setOnClickListener
            switchTab("recommended")
        }
        binding.tabLatest.setOnClickListener {
            if (currentTab == "latest") return@setOnClickListener
            switchTab("latest")
        }
        updateTabsUi()
    }

    private fun switchTab(tab: String) {
        currentTab = tab
        currentPage = 1
        hasMore = true
        binding.recyclerView.scrollToPosition(0)
        updateTabsUi()
        loadFeed(tab, 1, false)
    }

    private fun updateTabsUi() {
        val active = requireContext().getColor(R.color.accent_red)
        val inactive = requireContext().getColor(R.color.card_bg)
        binding.tabRecommended.setBackgroundColor(
            if (currentTab == "recommended") active else inactive
        )
        binding.tabLatest.setBackgroundColor(
            if (currentTab == "latest") active else inactive
        )
    }

    private fun setupSwipeRefresh() {
        binding.swipeRefresh.setOnRefreshListener {
            currentPage = 1
            hasMore = true
            loadFeed(currentTab, 1, false)
        }
    }

    private fun setupInfiniteScroll() {
        binding.recyclerView.addOnScrollListener(object : RecyclerView.OnScrollListener() {
            override fun onScrolled(recyclerView: RecyclerView, dx: Int, dy: Int) {
                super.onScrolled(recyclerView, dx, dy)
                if (dy <= 0) return
                if (isLoading || !hasMore) return

                val totalItems = layoutManager.itemCount
                val lastVisible = layoutManager.findLastVisibleItemPosition()

                if (lastVisible >= totalItems - 5) {
                    currentPage++
                    loadFeed(currentTab, currentPage, true)
                }
            }
        })
    }

    private fun loadFeed(tab: String, page: Int, append: Boolean) {
        if (isLoading) return
        isLoading = true

        if (!append) {
            binding.swipeRefresh.isRefreshing = true
        }

        lifecycleScope.launch {
            try {
                val response = ApiClient.service.getFeed(tab = tab, page = page, perPage = perPage)
                if (response.success) {
                    if (append) adapter.appendVideos(response.videos)
                    else adapter.setVideos(response.videos)
                    hasMore = response.hasMore
                } else {
                    hasMore = false
                }
            } catch (e: Exception) {
                Log.e("FeedFragment", "Ошибка загрузки", e)
                hasMore = false
            } finally {
                isLoading = false
                binding.swipeRefresh.isRefreshing = false
            }
        }
    }

    private fun openVideo(video: Video) {
        (requireActivity() as ru.asmr1.app.MainActivity).openPlayer(video)
    }

    override fun onDestroyView() {
        super.onDestroyView()
        _binding = null
    }
}

        notifyDataSetChanged()
    }

    override fun onCreateViewHolder(parent: ViewGroup, viewType: Int): VideoViewHolder {

            LayoutInflater.from(parent.context), parent, false
        )
        return VideoViewHolder(binding)

            binding.title.text = video.title

            val viewsText = formatViews(video.views)
            binding.meta.text = "${video.author.name} • $viewsText"

            if (video.thumbnail.isNotEmpty()) {
                binding.thumbnail.load(video.thumbnail) {
                    crossfade(true)

                }

                binding.duration.text = formatDuration(video.duration)
            } else {
                binding.duration.visibility = View.GONE
            }

        }
    }

            else -> "$n просмотров"
        }
    }

    private fun formatDuration(sec: Int): String {
        val m = sec / 60
        val s = sec % 60
        return "$m:${if (s < 10) "0$s" else s}"
    }

package ru.asmr1.app.utils

import java.text.SimpleDateFormat
import java.util.Date
import java.util.Locale
import java.util.TimeZone

fun formatViews(n: Int): String = when {
    n >= 1_000_000 -> String.format("%.1fM просмотров", n / 1_000_000.0)
    n >= 1_000 -> String.format("%.1fK просмотров", n / 1_000.0)
    else -> "$n просмотров"
}

fun formatDuration(sec: Int): String {
    val m = sec / 60
    val s = sec % 60
    return "$m:${if (s < 10) "0$s" else s}"
}

fun timeAgo(dateStr: String): String {
    if (dateStr.isEmpty()) return ""
    return try {
        val fmt = SimpleDateFormat("yyyy-MM-dd HH:mm:ss", Locale.US).apply {
            timeZone = TimeZone.getTimeZone("UTC")
        }
        val date = fmt.parse(dateStr) ?: return ""
        val diff = (System.currentTimeMillis() - date.time) / 1000
        when {
            diff < 60 -> "только что"
            diff < 3600 -> "${diff / 60} мин назад"
            diff < 86400 -> "${diff / 3600} ч назад"
            diff < 2592000 -> "${diff / 86400} дн назад"
            diff < 31536000 -> "${diff / 2592000} мес назад"
            else -> "${diff / 31536000} г назад"
        }
    } catch (_: Exception) {
        ""
    }
}
