package ru.asmr1.app.utils

import java.text.SimpleDateFormat
import java.util.Date
import java.util.Locale
import java.util.TimeZone

/**
 * 1234 → "1.2K просмотров"
 */
fun formatViews(n: Int): String = when {
    n >= 1_000_000 -> String.format("%.1fM просмотров", n / 1_000_000.0)
    n >= 1_000 -> String.format("%.1fK просмотров", n / 1_000.0)
    else -> "$n просмотров"
}

/**
 * Секунды → "12:34"
 */
fun formatDuration(sec: Int): String {
    val m = sec / 60
    val s = sec % 60
    return "$m:${if (s < 10) "0$s" else s}"
}

/**
 * "2025-09-26 12:34:56" → "5 мин назад"
 */
fun timeAgo(dateStr: String): String {
    if (dateStr.isEmpty()) return ""
    return try {
        val fmt = SimpleDateFormat("yyyy-MM-dd HH:mm:ss", Locale.US).apply {
            timeZone = TimeZone.getTimeZone("UTC")
        }
        val date = fmt.parse(dateStr) ?: return ""
        val diff = (System.currentTimeMillis() - date.time) / 1000
        when {
            diff < 60 -> "только что"
            diff < 3600 -> "${diff / 60} мин назад"
            diff < 86400 -> "${diff / 3600} ч назад"
            diff < 2592000 -> "${diff / 86400} дн назад"
            diff < 31536000 -> "${diff / 2592000} мес назад"
            else -> "${diff / 31536000} г назад"
        }
    } catch (_: Exception) {
        ""
    }
}

package ru.asmr1.app.ui

import android.view.LayoutInflater
import android.view.View
import android.view.ViewGroup
import androidx.recyclerview.widget.RecyclerView
import coil.load
import ru.asmr1.app.R
import ru.asmr1.app.data.Video
import ru.asmr1.app.databinding.ItemVideoBinding

class VideoAdapter(
    private var videos: List<Video> = emptyList(),
    private val onClick: (Video) -> Unit = {}
) : RecyclerView.Adapter<VideoAdapter.VideoViewHolder>() {

    fun setVideos(newVideos: List<Video>) {
        videos = newVideos
        notifyDataSetChanged()
    }

    fun appendVideos(newVideos: List<Video>) {
        val startPos = videos.size
        videos = videos + newVideos
        notifyItemRangeInserted(startPos, newVideos.size)
    }

    override fun onCreateViewHolder(parent: ViewGroup, viewType: Int): VideoViewHolder {
        val binding = ItemVideoBinding.inflate(
            LayoutInflater.from(parent.context), parent, false
        )
        return VideoViewHolder(binding)
    }

    override fun onBindViewHolder(holder: VideoViewHolder, position: Int) {
        holder.bind(videos[position])
    }

    override fun getItemCount(): Int = videos.size

    inner class VideoViewHolder(private val binding: ItemVideoBinding) :
        RecyclerView.ViewHolder(binding.root) {

        fun bind(video: Video) {
            // Название
            binding.title.text = video.title

            // Мета: автор • просмотры
            val viewsText = formatViews(video.views)
            binding.meta.text = "${video.author.name} • $viewsText"

            // Превью
            if (video.thumbnail.isNotEmpty()) {
                binding.thumbnail.load(video.thumbnail) {
                    crossfade(true)
                    placeholder(R.color.card_bg)
                }
            }

            // Аватар (круглый)
            if (video.author.avatar.isNotEmpty()) {
                binding.authorAvatar.load(video.author.avatar) {
                    crossfade(true)
                    placeholder(R.color.card_bg)
                }
            }

            // Длительность
            if (video.duration > 0) {
                binding.duration.visibility = View.VISIBLE
                binding.duration.text = formatDuration(video.duration)
            } else {
                binding.duration.visibility = View.GONE
            }

            // Клик
            binding.root.setOnClickListener { onClick(video) }
        }
    }

    private fun formatViews(n: Int): String {
        return when {
            n >= 1_000_000 -> String.format("%.1fM просмотров", n / 1_000_000.0)
            n >= 1_000 -> String.format("%.1fK просмотров", n / 1_000.0)
            else -> "$n просмотров"
        }
    }

    private fun formatDuration(sec: Int): String {
        val m = sec / 60
        val s = sec % 60
        return "$m:${if (s < 10) "0$s" else s}"
    }
}

import android.content.Intent

class FeedFragment : Fragment() {

    private var _binding: FragmentFeedBinding? = null
    private val binding get() = _binding!!

    private lateinit var adapter: VideoAdapter
    private lateinit var layoutManager: LinearLayoutManager

    private var currentTab = "recommended"
    private var currentPage = 1
    private var isLoading = false
    private var hasMore = true
    private val perPage = 20

    override fun onCreateView(
        inflater: LayoutInflater,
        container: ViewGroup?,
        savedInstanceState: Bundle?
    ): View {
        _binding = FragmentFeedBinding.inflate(inflater, container, false)
        return binding.root
    }

    override fun onViewCreated(view: View, savedInstanceState: Bundle?) {
        super.onViewCreated(view, savedInstanceState)

        setupRecyclerView()
        setupTabs()
        setupSwipeRefresh()
        setupInfiniteScroll()
        loadFeed(currentTab, 1, false)
    }

    private fun setupRecyclerView() {
        adapter = VideoAdapter { video ->
            openVideo(video)
        }
        layoutManager = LinearLayoutManager(requireContext())
        binding.recyclerView.layoutManager = layoutManager
        binding.recyclerView.adapter = adapter
    }

    private fun setupTabs() {
        binding.tabRecommended.setOnClickListener {
            if (currentTab == "recommended") return@setOnClickListener
            switchTab("recommended")
        }
        binding.tabLatest.setOnClickListener {
            if (currentTab == "latest") return@setOnClickListener
            switchTab("latest")
        }
        updateTabsUi()
    }

    private fun switchTab(tab: String) {
        currentTab = tab
        currentPage = 1
        hasMore = true
        binding.recyclerView.scrollToPosition(0)
        updateTabsUi()
        loadFeed(tab, 1, false)
    }

    private fun updateTabsUi() {
        val active = requireContext().getColor(R.color.accent_red)
        val inactive = requireContext().getColor(R.color.card_bg)
        binding.tabRecommended.setBackgroundColor(
            if (currentTab == "recommended") active else inactive
        )
        binding.tabLatest.setBackgroundColor(
            if (currentTab == "latest") active else inactive
        )
    }

    private fun setupSwipeRefresh() {
        binding.swipeRefresh.setOnRefreshListener {
            currentPage = 1
            hasMore = true
            loadFeed(currentTab, 1, false)
        }
    }

    private fun setupInfiniteScroll() {
        binding.recyclerView.addOnScrollListener(object : RecyclerView.OnScrollListener() {
            override fun onScrolled(recyclerView: RecyclerView, dx: Int, dy: Int) {
                super.onScrolled(recyclerView, dx, dy)
                if (dy <= 0) return
                if (isLoading || !hasMore) return

                val totalItems = layoutManager.itemCount
                val lastVisible = layoutManager.findLastVisibleItemPosition()

                if (lastVisible >= totalItems - 5) {
                    currentPage++
                    loadFeed(currentTab, currentPage, true)
                }
            }
        })
    }

    private fun loadFeed(tab: String, page: Int, append: Boolean) {
        if (isLoading) return
        isLoading = true

        if (!append) {
            binding.swipeRefresh.isRefreshing = true
        }

        lifecycleScope.launch {
            try {
                val response = ApiClient.service.getFeed(tab = tab, page = page, perPage = perPage)
                if (response.success) {
                    if (append) adapter.appendVideos(response.videos)
                    else adapter.setVideos(response.videos)
                    hasMore = response.hasMore
                } else {
                    hasMore = false
                }
            } catch (e: Exception) {
                Log.e("FeedFragment", "Ошибка загрузки", e)
                hasMore = false
            } finally {
                isLoading = false
                binding.swipeRefresh.isRefreshing = false
            }
        }
    }

    /**
     * Открываем нативный плеер.
     */
    private fun openVideo(video: Video) {
        (requireActivity() as ru.asmr1.app.MainActivity).openPlayer(video)
    }
        Log.d("FeedFragment", "🎬 Открываем видео: ${video.id} — ${video.title}")

        val intent = Intent(requireContext(), PlayerActivity::class.java).apply {
            putExtra(PlayerActivity.EXTRA_VIDEO_ID, video.id)
            putExtra(PlayerActivity.EXTRA_VIDEO_URL, video.videoUrl)
            putExtra(PlayerActivity.EXTRA_TITLE, video.title)
            putExtra(PlayerActivity.EXTRA_AUTHOR, video.author.name)
            putExtra(PlayerActivity.EXTRA_START_TIME_MS, 0L)
        }
        startActivity(intent)
    }

    override fun onDestroyView() {
        super.onDestroyView()
        _binding = null
    }
}

package ru.asmr1.app.player

import android.app.PictureInPictureParams
import android.content.res.Configuration
import android.os.Build
import android.os.Bundle
import android.util.Rational
import android.view.WindowManager
import android.widget.ImageButton
import androidx.appcompat.app.AppCompatActivity
import androidx.media3.common.MediaItem
import androidx.media3.common.Player
import androidx.media3.exoplayer.ExoPlayer
import androidx.media3.ui.PlayerView
import ru.asmr1.app.R

class PlayerActivity : AppCompatActivity() {

    companion object {
        const val EXTRA_VIDEO_URL = "video_url"
        const val EXTRA_VIDEO_ID = "video_id"
        const val EXTRA_TITLE = "title"
        const val EXTRA_AUTHOR = "author"
        const val EXTRA_START_TIME_MS = "start_time_ms"
    }

    private lateinit var player: ExoPlayer
    private lateinit var playerView: PlayerView
    private var videoId: Int = 0
    private var startTimeMs: Long = 0L

    override fun onCreate(savedInstanceState: Bundle?) {
        super.onCreate(savedInstanceState)
        setContentView(R.layout.activity_player)

        // Fullscreen
        window.setFlags(
            WindowManager.LayoutParams.FLAG_FULLSCREEN,
            WindowManager.LayoutParams.FLAG_FULLSCREEN
        )

        // Данные
        val videoUrl = intent.getStringExtra(EXTRA_VIDEO_URL) ?: ""
        videoId = intent.getIntExtra(EXTRA_VIDEO_ID, 0)
        startTimeMs = intent.getLongExtra(EXTRA_START_TIME_MS, 0L)

        // Кнопка закрытия
        findViewById<ImageButton>(R.id.btn_close).setOnClickListener {
            finish()
        }

        // ExoPlayer
        player = ExoPlayer.Builder(this).build()
        playerView = findViewById(R.id.player_view)
        playerView.player = player

        if (videoUrl.isNotEmpty()) {
            val mediaItem = MediaItem.fromUri(videoUrl)
            player.setMediaItem(mediaItem, startTimeMs)
            player.prepare()
            player.playWhenReady = true
        }

        // Auto-PiP: система сама переведёт в PiP при жесте Home (Android 12+)
        setupAutoPip()

        // Когда видео закончилось — закрываем
        player.addListener(object : Player.Listener {
            override fun onPlaybackStateChanged(state: Int) {
                if (state == Player.STATE_ENDED) {
                    finish()
                }
            }
        })
    }

    /**
     * ★ ГЛАВНОЕ: включаем авто-PiP.
     * Работает на Android 12+ (включая HyperOS, MIUI).
     */
    private fun setupAutoPip() {
        if (Build.VERSION.SDK_INT >= Build.VERSION_CODES.S) {
            try {
                val params = PictureInPictureParams.Builder()
                    .setAspectRatio(Rational(16, 9))
                    .setAutoEnterEnabled(true)
                    .setSeamlessResizeEnabled(true)
                    .build()
                setPictureInPictureParams(params)
            } catch (e: Exception) {
                // ignore
            }
        }
    }

    override fun onPictureInPictureModeChanged(
        isInPictureInPictureMode: Boolean,
        newConfig: Configuration
    ) {
        super.onPictureInPictureModeChanged(isInPictureInPictureMode, newConfig)
        // Скрываем кнопку закрытия в PiP
        findViewById<ImageButton>(R.id.btn_close).visibility =
            if (isInPictureInPictureMode) android.view.View.GONE else android.view.View.VISIBLE
    }

    override fun onStop() {
        super.onStop()
        // Паузим если уходим в фон (кроме PiP)
        if (Build.VERSION.SDK_INT >= Build.VERSION_CODES.N && !isInPictureInPictureMode) {
            player.pause()
        }
    }

    override fun onDestroy() {
        super.onDestroy()
        player.release()
    }
}

package ru.asmr1.app.player

import android.os.Bundle

class PlayerFragment : Fragment() {

    companion object {
        const val ARG_VIDEO_ID = "video_id"
        const val ARG_VIDEO_URL = "video_url"
        const val ARG_TITLE = "title"
        const val ARG_AUTHOR_ID = "author_id"
        const val ARG_AUTHOR_NAME = "author_name"
        const val ARG_AUTHOR_AVATAR = "author_avatar"
        const val ARG_VIEWS = "views"
        const val ARG_PUBLISHED = "published"
        const val ARG_DESCRIPTION = "description"
        const val ARG_THUMBNAIL = "thumbnail"

        fun newInstance(video: Video): PlayerFragment {
            return PlayerFragment().apply {
                arguments = Bundle().apply {
                    putInt(ARG_VIDEO_ID, video.id)
                    putString(ARG_VIDEO_URL, video.videoUrl)
                    putString(ARG_TITLE, video.title)
                    putInt(ARG_AUTHOR_ID, video.author.id)
                    putString(ARG_AUTHOR_NAME, video.author.name)
                    putString(ARG_AUTHOR_AVATAR, video.author.avatar)
                    putInt(ARG_VIEWS, video.views)
                    putString(ARG_PUBLISHED, video.published)
                    putString(ARG_DESCRIPTION, video.description)
                    putString(ARG_THUMBNAIL, video.thumbnail)
                }
            }
        }
    }

    private var _binding: FragmentPlayerBinding? = null
    private val binding get() = _binding!!

    private var player: ExoPlayer? = null

    override fun onCreateView(
        inflater: LayoutInflater,
        container: ViewGroup?,
        savedInstanceState: Bundle?
    ): View {
        _binding = FragmentPlayerBinding.inflate(inflater, container, false)
        return binding.root
    }

    override fun onViewCreated(view: View, savedInstanceState: Bundle?) {
        super.onViewCreated(view, savedInstanceState)

        val args = requireArguments()
        val videoUrl = args.getString(ARG_VIDEO_URL) ?: ""
        val title = args.getString(ARG_TITLE) ?: ""
        val authorName = args.getString(ARG_AUTHOR_NAME) ?: ""
        val authorAvatar = args.getString(ARG_AUTHOR_AVATAR) ?: ""
        val views = args.getInt(ARG_VIEWS)
        val published = args.getString(ARG_PUBLISHED) ?: ""
        val description = args.getString(ARG_DESCRIPTION) ?: ""
        val thumbnail = args.getString(ARG_THUMBNAIL) ?: ""

        // Заполняем UI
        binding.title.text = title
        binding.authorName.text = authorName
        binding.meta.text = "${formatViews(views)} • ${timeAgo(published)}"
        binding.description.text = description

        if (authorAvatar.isNotEmpty()) {
            binding.authorAvatar.load(authorAvatar)
        }

        if (thumbnail.isNotEmpty() && videoUrl.isEmpty()) {
            binding.thumbnail.load(thumbnail)
        }

        // Кнопка закрытия
        binding.btnClose.setOnClickListener {
            parentFragmentManager.popBackStack()
        }

        // ExoPlayer
        player = ExoPlayer.Builder(requireContext()).build().also { exo ->
            binding.playerView.player = exo

            if (videoUrl.isNotEmpty()) {
                val mediaItem = MediaItem.fromUri(videoUrl)
                exo.setMediaItem(mediaItem)
                exo.prepare()
                exo.playWhenReady = true
            }

            exo.addListener(object : Player.Listener {
                override fun onPlaybackStateChanged(state: Int) {
                    if (state == Player.STATE_ENDED) {
                        parentFragmentManager.popBackStack()
                    }
                }
            })
        }
    }

    override fun onStart() {
        super.onStart()
        player?.play()
    }

    override fun onStop() {
        super.onStop()
        // Паузим если не в PiP
        val activity = requireActivity()
        if (android.os.Build.VERSION.SDK_INT < android.os.Build.VERSION_CODES.N ||
            !activity.isInPictureInPictureMode) {
            player?.pause()
        }
    }

    override fun onDestroyView() {
        super.onDestroyView()
        player?.release()
        player = null
        _binding = null
    }

    fun getPlayer(): ExoPlayer? = player
}
